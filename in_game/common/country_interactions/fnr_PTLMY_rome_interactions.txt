# The Ptolemaic Kingdom and Rome (Pax Romana)
# The Ptolemies seek Rome's friendship or a royal marriage; Rome asks for the grain fleet.

# Seek Roman Amicitia - Friend and Ally of the Roman People (25 years, both sides)
fnr_PTLMY_seek_amicitia = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS

	potential = {
		scope:actor = { has_or_had_tag = PTLMY }
	}

	allow = {
		scope:actor = {
			num_of_diplomats >= 1
		}
	}

	cooldown = { type = fnr_PTLMY_seek_amicitia years = 10 }

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info
		}
		visible = {
			fnr_PTLMY_is_rome = yes
		}
		enabled = {
			NOT = { is_at_war_with = scope:actor }
		}
	}

	diplo_chance = {
		base = -20
		opinion = 0.5
		trust_in_actor = 0.5
	}

	accept = {
		add = {
			desc = "fnr_PTLMY_reason_grain"
			value = 15
		}
		add = {
			desc = "fnr_PTLMY_reason_already_friends"
			if = {
				limit = { has_country_modifier = fnr_PTLMY_egyptian_friendship }
				value = 20
			}
		}
	}

	effect = {
		scope:actor = {
			add_country_modifier = { modifier = fnr_PTLMY_amicus_populi_romani years = 25 mode = replace }
			add_opinion = { target = scope:recipient modifier = fnr_PTLMY_roman_amicitia }
		}
		scope:recipient = {
			add_country_modifier = { modifier = fnr_PTLMY_egyptian_friendship years = 25 mode = replace }
			add_opinion = { target = scope:actor modifier = fnr_PTLMY_roman_amicitia }
		}
	}

	reject_effect = {
		scope:actor = {
			add_opinion = { target = scope:recipient modifier = fnr_PTLMY_spurned_by_rome }
		}
	}

	ai_will_do = {
		value = 50
	}
}

# A Queen for Rome - the Lagid and Roman houses united (30 years, both sides). Our ruler must be a queen.
fnr_PTLMY_royal_marriage_rome = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS

	potential = {
		scope:actor = { has_or_had_tag = PTLMY }
	}

	allow = {
		scope:actor = {
			num_of_diplomats >= 1
			ruler ?= { is_female = yes }
		}
	}

	cooldown = { type = fnr_PTLMY_royal_marriage_rome years = 20 }

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info
		}
		visible = {
			fnr_PTLMY_is_rome = yes
		}
		enabled = {
			NOT = { is_at_war_with = scope:actor }
		}
	}

	diplo_chance = {
		base = -10
		opinion = 0.5
		trust_in_actor = 0.5
	}

	accept = {
		add = {
			desc = "fnr_PTLMY_reason_amicitia"
			if = {
				limit = { scope:actor = { has_country_modifier = fnr_PTLMY_amicus_populi_romani } }
				value = 25
			}
		}
		add = {
			desc = "fnr_PTLMY_reason_egypt_wealth"
			value = 10
		}
	}

	effect = {
		scope:actor = {
			add_country_modifier = { modifier = fnr_PTLMY_lagid_roman_union years = 30 mode = replace }
			add_opinion = { target = scope:recipient modifier = fnr_PTLMY_lagid_marriage }
		}
		scope:recipient = {
			add_country_modifier = { modifier = fnr_PTLMY_lagid_roman_union years = 30 mode = replace }
			add_opinion = { target = scope:actor modifier = fnr_PTLMY_lagid_marriage }
		}
	}

	reject_effect = {
		scope:actor = {
			add_prestige = prestige_weak_penalty
		}
	}

	ai_will_do = {
		value = 30
	}
}

# Request the Grain Fleet - Rome pays 100 gold for The Grain Fleet (10 years)
fnr_PTLMY_request_grain_fleet = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS

	potential = {
		scope:actor = { fnr_PTLMY_is_rome = yes }
	}

	allow = {
		scope:actor = {
			num_of_diplomats >= 1
			gold >= 100
		}
	}

	cooldown = { type = fnr_PTLMY_request_grain_fleet years = 10 }

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info
		}
		visible = {
			has_or_had_tag = PTLMY
		}
		enabled = {
			NOT = { is_at_war_with = scope:actor }
		}
	}

	diplo_chance = {
		base = 0
		opinion = 0.5
		trust_in_actor = 0.5
	}

	accept = {
		add = {
			desc = "fnr_PTLMY_reason_payment"
			value = 20
		}
		add = {
			desc = "fnr_PTLMY_reason_bad_harvest"
			if = {
				limit = {
					scope:recipient = {
						OR = {
							has_country_modifier = fnr_PTLMY_nile_low
							has_country_modifier = fnr_PTLMY_nile_failed
							has_country_modifier = fnr_PTLMY_nile_granaries_opened
						}
					}
				}
				value = -50
			}
		}
	}

	effect = {
		scope:actor = {
			add_gold = -100
			add_country_modifier = { modifier = fnr_PTLMY_grain_fleet years = 10 mode = replace }
		}
		scope:recipient = {
			add_gold = 100
			add_opinion = { target = scope:actor modifier = fnr_PTLMY_grain_for_rome }
		}
	}

	reject_effect = {
		scope:actor = {
			add_opinion = { target = scope:recipient modifier = fnr_PTLMY_refused_rome }
		}
	}

	ai_will_do = {
		value = 50
	}
}

# Restore the Diocese of Italia - 60% of italy_region
fnr_ROM_restore_diocese_italia = {
	decision_category = country_specific_decisions

	tags = { ROM }

	potential = {
		has_or_had_tag = ROM
		NOT = { has_variable = fnr_ROM_diocese_italia }
	}

	allow = {
		region:italy_region = {
			any_location_in_region = {
				percent >= 0.6
				owner = root
			}
		}
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_ROM_restore_diocese_italia.a
		ai_chance = { value = 100 }
		effect = {
			fnr_ROM_restore_diocese_effect = { KEY = italia }
		}
	}
}

# Pay a Donative - gold for the legions, every 5 years
fnr_ROM_pay_donative = {
	decision_category = country_specific_decisions

	tags = { ROM }

	potential = {
		has_or_had_tag = ROM
	}

	allow = {
		gold >= 250
		NOT = { has_variable = fnr_ROM_donative_cooldown }
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:peasants_estate }
	}

	option = {
		name = fnr_ROM_pay_donative.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -250
			fnr_ROM_change_loyalty_effect = { VALUE = 15 TT = plus_15 }
			add_country_modifier = { modifier = fnr_ROM_donative years = 5 mode = add_and_extend }
			set_variable = { name = fnr_ROM_donative_cooldown years = 5 }
		}
	}
}

# ==================== Client kingdoms (one each) ====================
# Each carves a vassal kingdom out of our lands in its historical areas; it joins the Foederati.

# Establish the client kingdom of Regnum Armeniae
fnr_ROM_client_armenia = {
	decision_category = country_specific_decisions

	tags = { ROM }

	potential = {
		has_or_had_tag = ROM
		NOT = { has_variable = fnr_ROM_client_armenia }
		any_owned_location = {
			fnr_ROM_client_area_armenia = yes
		}
	}

	allow = {
		at_war = no
		gold >= 100
		custom_tooltip = {
			text = fnr_ROM_client_enough_land_tt
			any_owned_location = {
				count >= 3
				is_capital = no
				fnr_ROM_client_area_armenia = yes
			}
		}
		custom_tooltip = {
			text = fnr_ROM_client_not_capital_tt
			capital = { fnr_ROM_client_area_armenia = no }
		}
	}

	ai_will_do = { value = 0 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_ROM_client_armenia.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -100
			fnr_ROM_create_client_effect = { KEY = armenia }
		}
	}
}

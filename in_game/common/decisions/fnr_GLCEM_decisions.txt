# Gallic Empire Decisions (Pax Romana)
# Replace the orphaned flavor_GLCEM.2 / .3 formation events: Tarraconensis and Britannia,
# the historical provinces of the Gallic Empire, can be released as loyal vassals at any time.

# Establish Tarraconensis - hands our Iberian lands to a new vassal kingdom
fnr_GLCEM_establish_tarraconensis = {
	decision_category = country_specific_decisions

	tags = { GLCEM }

	potential = {
		NOT = { exists = c:TRCNS }
		any_owned_location = { fnr_GLCEM_is_tarraconensis_location = yes }
	}

	allow = {
		at_war = no
		any_owned_location = {
			count >= 12
			is_capital = no
			fnr_GLCEM_is_tarraconensis_location = yes
		}
	}

	ai_will_do = { value = 0 } # the AI would hand away whole regions

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = fnr_GLCEM_establish_tarraconensis.a
		ai_chance = { value = 100 }
		effect = {
			custom_tooltip = fnr_GLCEM_establish_tarraconensis_tt
			custom_tooltip = fnr_GLCEM_released_lock_tt
			hidden_effect = {
				fnr_GLCEM_release_tarraconensis_effect = yes
			}
			add_prestige = prestige_mild_bonus
		}
	}
}

# ==================== Holy sites ====================

# Holy site - the Forest of the Carnutes: permanent location upgrade; country bonus while owned (yearly pulse)
fnr_GLCEM_restore_site_chartres = {
	decision_category = religious_decisions

	tags = { GLCEM }

	potential = {
		has_variable = fnr_GLCEM_path_druidic
		owns = location:chartres
		NOT = { location:chartres = { has_location_modifier = fnr_GLCEM_restored_site } }
	}

	allow = {
		owns = location:chartres
		gold >= 300
		at_war = no
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	option = {
		name = fnr_GLCEM_restore_site_chartres.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -300
			location:chartres = {
				add_location_modifier = { modifier = fnr_GLCEM_restored_site years = -1 }
			}
			add_country_modifier = { modifier = fnr_GLCEM_site_chartres years = -1 }
			add_prestige = prestige_weak_bonus
			custom_tooltip = fnr_GLCEM_restored_site_tt
		}
	}
}

# Grant Tarraconensis the right to reshape itself - lifts the formation lock on our released vassal

	decision_category = country_specific_decisions

	tags = { GLCEM }

	potential = {
		exists = c:TRCNS

			is_subject_of = root
			has_variable = fnr_formation_locked
		}
	}

	allow = {
		at_war = no
	}

	ai_will_do = { value = 0 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

	option = {
		name = fnr_GLCEM_grant_tarraconensis.a

			c:TRCNS = {
				remove_variable = fnr_formation_locked
				add_opinion = { target = root modifier = fnr_GLCEM_granted_self_rule }
			}
			custom_tooltip = fnr_GLCEM_grant_self_rule_tt
		}
	}
}

# The Great Assembly - peacetime league action every 5 years
fnr_GLCEM_great_assembly = {
	decision_category = country_specific_decisions

	tags = { GLCEM }

	potential = {
		exists = international_organization:fnr_GLCEM_celtic_league_io
	}

	allow = {
		at_war = no
		gold >= 100
		NOT = { has_variable = fnr_GLCEM_assembly_cooldown }
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_GLCEM_great_assembly.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -100
			add_prestige = prestige_mild_bonus
			international_organization:fnr_GLCEM_celtic_league_io = {
				every_international_organization_member = {
					add_country_modifier = { modifier = fnr_GLCEM_great_assembly years = 5 mode = add_and_extend }
					if = {
						limit = { NOT = { tag = GLCEM } }
						add_opinion = { target = root modifier = fnr_GLCEM_attended_assembly }
					}
				}
			}
			set_variable = { name = fnr_GLCEM_assembly_cooldown years = 5 }
			custom_tooltip = fnr_GLCEM_great_assembly_tt
		}
	}
}

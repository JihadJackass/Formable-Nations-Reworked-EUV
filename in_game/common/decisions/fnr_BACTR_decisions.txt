# Bactria and the Bactrian Synod
# The Synod is proclaimed by decision (flavor_BACTR.3). Its leader restores the old kingdoms
# of Alexander's empire as client kingdoms, then claims the Macedonian Empire (flavor_BACTR.9).

# Proclaim the Bactrian Synod
fnr_BACTR_proclaim_synod = {
	decision_category = country_specific_decisions

	tags = { BACTR }

	potential = {
		has_or_had_tag = BACTR
		NOT = { exists = international_organization:fnr_bactria_synod_io }
	}

	allow = {
		religion = religion:fnr_greco_buddhism_religion
		at_war = no
		gold >= 200
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:crown_estate }
	}

	option = {
		name = fnr_BACTR_proclaim_synod.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -200
			trigger_event_non_silently = flavor_BACTR.3
		}
	}
}

# Restore ARMNA (armenia) - flavor_BACTR.4
fnr_BACTR_restore_armenia = {
	decision_category = country_specific_decisions

	tags = { BACTR }

	potential = {
		has_or_had_tag = BACTR
		fnr_BACTR_leads_synod = yes
		NOT = { has_variable = fnr_BACTR_client_armenia }
		NOT = { has_variable = fnr_BACTR_client_pending_armenia }
		NOT = { tag_exists = ARMNA }
		any_owned_location = { fnr_BACTR_client_area_armenia = yes }
	}

	allow = {
		at_war = no
		gold >= 100

		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_BACTR_restore_armenia.a

			set_variable = { name = fnr_BACTR_client_pending_armenia days = 30 }
			trigger_event_non_silently = flavor_BACTR.4
		}
	}
}

# Heir of Alexander - the Macedonian Empire (flavor_BACTR.9)
fnr_BACTR_heir_of_alexander = {
	decision_category = country_specific_decisions

	tags = { BACTR }

	potential = {
		has_or_had_tag = BACTR
		fnr_BACTR_leads_synod = yes
		NOT = { has_variable = fnr_BACTR_heir_of_alexander }
		NOT = { has_variable = fnr_BACTR_heir_pending }
	}

	allow = {
		at_war = no
		gold >= 500
		custom_tooltip = {
			text = fnr_BACTR_macedonia_restored_tt
			has_variable = fnr_BACTR_client_macedonia
		}
		custom_tooltip = {
			text = fnr_BACTR_four_kingdoms_tt
			has_variable = fnr_BACTR_clients
			var:fnr_BACTR_clients >= 4
		}
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_BACTR_heir_of_alexander.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -500
			set_variable = { name = fnr_BACTR_heir_pending days = 30 }
			trigger_event_non_silently = flavor_BACTR.9
		}
	}
}

# Decisions for the Roman provinces of the Gallic Empire (Pax Romana)

# Send Auxiliaries to the Emperor - our overlord gets manpower and morale for 5 years
fnr_PROV_send_auxiliaries = {
	decision_category = country_specific_decisions

	potential = {
		fnr_PROV_is_province = yes
		is_subject = yes
	}

	allow = {
		NOT = { has_country_modifier = fnr_PROV_auxiliaries_sent }
		gold >= 50
	}

	ai_will_do = { value = 30 }

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:crown_estate }
	}

	option = {
		name = fnr_PROV_send_auxiliaries.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -50
			add_manpower = -1
			add_country_modifier = { modifier = fnr_PROV_auxiliaries_sent years = 10 }
			overlord ?= {
				add_country_modifier = { modifier = fnr_PROV_provincial_auxiliaries years = 5 mode = add_and_extend }
				add_opinion = { modifier = fnr_PROV_loyal_province target = root }
			}
		}
	}
}

# Remit the Provincial Tribute - 100 gold to our overlord
fnr_PROV_remit_tribute = {
	decision_category = country_specific_decisions

	potential = {
		fnr_PROV_is_province = yes
		is_subject = yes
	}

	allow = {
		NOT = { has_country_modifier = fnr_PROV_tribute_sent }
		gold >= 100
	}

	ai_will_do = { value = 20 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:crown_estate }
	}

	option = {
		name = fnr_PROV_remit_tribute.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -100
			add_country_modifier = { modifier = fnr_PROV_tribute_sent years = 5 }
			overlord ?= {
				add_gold = 100
				add_opinion = { modifier = fnr_PROV_loyal_province target = root }
			}
		}
	}
}

# Throw Off the Yoke - the rare Provincial Rising. Hard requirements, once per game.
fnr_PROV_throw_off_the_yoke = {
	decision_category = country_specific_decisions

	potential = {
		fnr_PROV_is_province = yes
		is_subject = yes
		NOT = { has_variable = fnr_PROV_rising_done }
		fnr_PROV_years_at_least = { VALUE = 15 }
	}

	allow = {
		fnr_PROV_years_at_least = { VALUE = 25 }
		stability >= 50
		gold >= 300
		fnr_PROV_overlord_weakened = yes
		at_war = no
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
	}

	option = {
		name = fnr_PROV_throw_off_the_yoke.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -300
			trigger_event_non_silently = fnr_PROV.3
		}
	}
}

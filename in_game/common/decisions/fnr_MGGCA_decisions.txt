# The Concord of the Two Peoples (every 10 years)
fnr_MGGCA_concord = {
	decision_category = country_specific_decisions

	potential = {
		fnr_MGGCA_can_revive = yes
		fnr_MGGCA_stage_at_least = { VALUE = 1 }
	}

	allow = {
		gold >= 150
		NOT = { has_variable = fnr_MGGCA_concord }
	}

	ai_will_do = { value = 10 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = fnr_MGGCA_concord.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -150
			set_variable = { name = fnr_MGGCA_concord years = 10 }
			if = {
				limit = { has_country_modifier = fnr_MGGCA_italian_resentment }
				remove_country_modifier = fnr_MGGCA_italian_resentment
			}
			add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
			add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
			add_stability = stability_weak_bonus
		}
	}
}

# Ptolemaic Kingdom Decisions (Pax Romana)

# ==================== The Hellenized Egypt path (any Egyptian state) ====================

# Invite the Greek Scholars - step 1 of the Hellenized Egypt path
fnr_PTLMY_invite_greek_scholars = {
	decision_category = country_specific_decisions

	potential = {
		fnr_PTLMY_allowed = yes
		NOT = { tag_exists = PTLMY }
		capital ?= { region = region:egypt_region }
		fnr_PTLMY_is_hellenic = no
		NOT = { has_variable = fnr_PTLMY_hellenizing }
	}

	allow = {
		owns = location:alexandria
		gold >= 150
	}

	ai_will_do = { value = 5 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }
	}

	option = {
		name = fnr_PTLMY_invite_greek_scholars.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -150
			set_variable = fnr_PTLMY_hellenizing
			set_variable = { name = fnr_PTLMY_hellenizing_wait years = 5 }
			add_country_modifier = { modifier = fnr_PTLMY_greek_scholars years = 10 mode = add_and_extend }
			custom_tooltip = fnr_PTLMY_invite_greek_scholars_tt
		}
	}
}

# A Greek Court at Alexandria - step 2: we may now form the Ptolemaic Kingdom
fnr_PTLMY_greek_court = {
	decision_category = country_specific_decisions

	potential = {
		fnr_PTLMY_allowed = yes
		NOT = { tag_exists = PTLMY }
		has_variable = fnr_PTLMY_hellenizing
		NOT = { has_variable = fnr_PTLMY_hellenized }
	}

	allow = {
		NOT = { has_variable = fnr_PTLMY_hellenizing_wait }
		owns = location:alexandria
		gold >= 200
		stability >= 0
	}

	ai_will_do = { value = 5 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:clergy_estate }
	}

	option = {
		name = fnr_PTLMY_greek_court.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -200
			set_variable = fnr_PTLMY_hellenized
			add_accepted_culture = culture:greek_culture
			add_stability = stability_mild_penalty
			add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
			custom_tooltip = fnr_PTLMY_greek_court_tt
		}
	}
}

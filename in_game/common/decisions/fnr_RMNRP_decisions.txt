# Roman Republic Decisions (Pax Romana)

# Appoint a Dictator - wartime emergency powers, at most 10 years. Ends with fnr_RMNRP.6 (resign or refuse).
fnr_RMNRP_appoint_dictator = {
	decision_category = country_specific_decisions

	tags = { RMNRP ROM }

	potential = {
		has_or_had_tag = RMNRP
		government_type = government_type:republic
		NOT = { has_variable = fnr_RMNRP_dictator_active }
	}

	allow = {
		at_war = yes
		NOT = { has_variable = fnr_RMNRP_dictator_cooldown }
	}

	ai_will_do = { value = 20 }

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_RMNRP_appoint_dictator.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = fnr_RMNRP_dictator_active
			add_country_modifier = { modifier = fnr_RMNRP_dictatorship years = 10 }
			fnr_RMNRP_change_senate_effect = { VALUE = -15 }
			custom_tooltip = fnr_RMNRP_appoint_dictator_tt
		}
	}
}

# Restore Romanitas - the Republic chooses how Roman it means to be.
#  A: Roman becomes our primary culture (our old culture stays accepted) - like vanilla's change_primary_culture
#  B: Roman becomes an accepted culture only; we can still Become Roman later
# Either way a quarter of Rome's people take up Roman culture and the Romanization cabinet action unlocks.
fnr_RMNRP_restore_romanitas = {
	decision_category = country_specific_decisions

	tags = { RMNRP ROM }

	potential = {
		has_or_had_tag = RMNRP
		NOT = { has_variable = fnr_RMNRP_romanitas }
	}

	allow = {
		owns = location:rome
		has_advance = RMNRP_mos_maiorum
		gold >= 200
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
	}

	option = {
		name = fnr_RMNRP_restore_romanitas.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -200
			set_variable = fnr_RMNRP_romanitas
			fnr_RMNRP_become_roman_effect = yes
			add_stability = stability_mild_penalty
			fnr_RMNRP_seed_rome_effect = yes
			custom_tooltip = fnr_RMNRP_romanitas_unlock_tt
		}
	}

	option = {
		name = fnr_RMNRP_restore_romanitas.b
		ai_chance = { value = 0 }
		effect = {
			add_gold = -200
			set_variable = fnr_RMNRP_romanitas
			if = {
				limit = {
					NOT = { culture = culture:roman_culture }
					NOT = { has_accepted_culture = culture:roman_culture }
				}
				add_accepted_culture = culture:roman_culture
			}
			add_prestige = prestige_mild_bonus
			fnr_RMNRP_seed_rome_effect = yes
			custom_tooltip = fnr_RMNRP_romanitas_unlock_tt
		}
	}
}

# Proclaim the Great Republic - the Republic becomes Rome without an emperor: the Roman Empire's
# dioceses and legions, ruled by a smaller Senate of generals and scholars (fnr_ROM_senate_of_generals).
# The name is chosen in fnr_ROM.1.
fnr_RMNRP_proclaim_great_republic = {
	decision_category = country_specific_decisions

	tags = { RMNRP }

	potential = {
		tag = RMNRP
		government_type = government_type:republic
		NOT = { has_variable = fnr_RMNRP_imperial_path }
	}

	allow = {
		owns = location:rome
		region:italy_region = {
			any_location_in_region = {
				percent >= 0.8
				owner ?= root
			}
		}
		country_rank = country_rank:rank_empire
		trigger_if = {
			limit = { has_variable = fnr_RMNRP_senate_authority }
			var:fnr_RMNRP_senate_authority >= 60
		}
		trigger_else = { always = no }
		republican_tradition >= 50
		NOT = { country_exists = c:BYZ }
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = fnr_RMNRP_proclaim_great_republic.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = fnr_RMNRP_great_republic
			custom_tooltip = fnr_RMNRP_great_republic_tt
			form_country = formable_country:ROM_BYZ_f
		}
	}
}

	# age_4_reformation

		# Capstone - Res Publica Aeterna (fires fnr_RMNRP.10 through the yearly pulse)
		RMNRP_res_publica_aeterna = {
			age = age_4_reformation
			icon = imperial_ambitions
			potential = {
				NOT = { has_variable = fnr_RMNRP_imperial_path } # an empire cannot claim the eternal republic
				has_or_had_tag = RMNRP
			}
			allow = {
				government_type = government_type:republic
				trigger_if = {
					limit = { has_variable = fnr_RMNRP_senate_authority }
					var:fnr_RMNRP_senate_authority >= 60
				}
				trigger_else = { always = no }
				republican_tradition >= 75
				region:italy_region = {
					any_location_in_region = {
						percent >= 0.8
						owner ?= root
					}
				}
			}

			monthly_prestige = 0.1
			monthly_republican_tradition = 0.1
			research_speed_modifier = 0.05

			requires = confessionalism_advance
			allow_children = no
		}

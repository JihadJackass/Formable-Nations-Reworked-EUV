on_game_start = {
	on_actions = {
		fnr_on_game_start_actions
	}
}

fnr_on_game_start_actions = {
	effect = {
		# FNR - Historical formables enabled by default; every country starts in sync
		every_country = {
			set_variable = {
				name = fnr_historical_enabled
				value = yes
			}
			set_variable = fnr_toggles_synced
		}

		# Papal States - set variable for GUI visibility
		every_country = {
			limit = {
				has_reform = government_reform:papacy_reform
				tag = PAP
			}
			set_variable = fnr_is_papacy
		}
	}
}

# New countries copy every formable toggle (categories and individual formables)
on_new_country_formed = {
	on_actions = {
		fnr_on_new_country_formed
	}
}

fnr_on_new_country_formed = {
	effect = {
		fnr_sync_formable_toggles_effect = yes
		# Special: Papacy toggle
		if = {
			limit = {
				has_reform = government_reform:papacy_reform
				tag = PAP
				any_country = { has_variable = fnr_is_papacy }
			}
			set_variable = { name = fnr_is_papacy value = yes }
		}
	}
}

on_released_country = {
	on_actions = {
		fnr_on_released_country
	}
}

fnr_on_released_country = {
	effect = {
		fnr_sync_formable_toggles_effect = yes
	}
}

# Safety net: countries created some other way (event-created kingdoms, rebels winning) sync within a year
yearly_country_pulse = {
	on_actions = {
		fnr_toggle_sync_pulse
	}
}

fnr_toggle_sync_pulse = {
	trigger = {
		NOT = { has_variable = fnr_toggles_synced }
	}
	effect = {
		fnr_sync_formable_toggles_effect = yes
	}
}

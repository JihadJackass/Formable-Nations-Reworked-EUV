# Ptolemaic Kingdom pulses (Pax Romana)

yearly_country_pulse = {
	on_actions = {
		fnr_PTLMY_yearly_pulse
		fnr_PTLMY_yearly_crown_pulse
		fnr_PTLMY_yearly_crown_pharaoh_pulse
		fnr_PTLMY_yearly_flavour_pulse
		fnr_PTLMY_yearly_serapis_pulse
		fnr_PTLMY_yearly_rome_pulse
	}
}

# Upkeep: the crown, the Library, the Nile, the League, Library milestones
fnr_PTLMY_yearly_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
	}
	effect = {
		fnr_PTLMY_yearly_upkeep_effect = yes
	}
}

# The crown at its extremes - ~15% a year (each event has its own 10-year cooldown)
fnr_PTLMY_yearly_crown_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
		fnr_PTLMY_crown_at_least = { VALUE = 60 }
	}
	random_events = {
		chance_to_happen = 15
		100 = fnr_PTLMY.10	# Unrest in the Thebaid (Basileus)
	}
}

fnr_PTLMY_yearly_crown_pharaoh_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
		fnr_PTLMY_crown_below = { VALUE = -59 }
	}
	random_events = {
		chance_to_happen = 15
		100 = fnr_PTLMY.11	# The Alexandrian Mob (Pharaoh)
	}
}

# Flavour - ~10% a year
fnr_PTLMY_yearly_flavour_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
	}
	random_events = {
		chance_to_happen = 7
		100 = fnr_PTLMY.25	# A Scholar at the Harbour
	}
}

fnr_PTLMY_yearly_serapis_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
	}
	random_events = {
		chance_to_happen = 4
		100 = fnr_PTLMY.12	# The Cult of Serapis (once)
	}
}

# Rome: a Ptolemaic subject of Rome may become Aegyptus - ~3% a year, asks the Roman overlord
fnr_PTLMY_yearly_rome_pulse = {
	trigger = {
		has_or_had_tag = PTLMY
		overlord ?= { fnr_PTLMY_is_rome = yes }
	}
	random_events = {
		chance_to_happen = 3
		100 = fnr_PTLMY.51
	}
}

# A new ruler: the epithet, and maybe a rival sibling
on_new_ruler = {
	on_actions = {
		fnr_PTLMY_on_new_ruler
		fnr_PTLMY_on_new_ruler_rival
	}
}

fnr_PTLMY_on_new_ruler = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
	}
	events = {
		fnr_PTLMY.40
	}
}

fnr_PTLMY_on_new_ruler_rival = {
	trigger = {
		has_or_had_tag = PTLMY
		has_variable = fnr_PTLMY_formed
	}
	random_events = {
		chance_to_happen = 25
		100 = fnr_PTLMY.41
	}
}

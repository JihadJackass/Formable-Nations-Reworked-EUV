# Rhineland monthly pulse
# Counts months at war from the Age of Absolutism on, for 'Learning from our experience' (flavor_RHLND.2, 120 months)

monthly_country_pulse = {
	on_actions = {
		fnr_RHLND_monthly_pulse
	}
}

fnr_RHLND_monthly_pulse = {
	trigger = {
		tag = RHLND
		at_war = yes
		current_age_or_later = { age = age_5_absolutism }
		NOT = { has_fired_unique_event = flavor_RHLND.2 }
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = fnr_RHLND_war_months } }
			set_variable = { name = fnr_RHLND_war_months value = 0 }
		}
		change_variable = { name = fnr_RHLND_war_months add = 1 }
	}
}

# Magna Graecia yearly pulses (Pax Romana)
# Italian Discontent grows likelier with each stage of the Hellenic Revival (about 8% a year per stage).

yearly_country_pulse = {
	on_actions = {
		fnr_MGGCA_discontent_stage_1
		fnr_MGGCA_discontent_stage_2
		fnr_MGGCA_discontent_stage_3
		fnr_MGGCA_discontent_stage_4
		fnr_MGGCA_foreign_pulse
	}
}

fnr_MGGCA_discontent_stage_1 = {
	trigger = {
		fnr_MGGCA_can_revive = yes
		fnr_MGGCA_stage_at_least = { VALUE = 1 }
	}
	random_events = {
		chance_to_happen = 8

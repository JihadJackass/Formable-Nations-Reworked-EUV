fnr_ERSHR_zoroastrian_reaction = {

	image = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	monthly_spawn_chance = monthly_spawn_chance_high

	can_start = {
		has_any_active_disaster = no
		is_subject = no
		religion = religion:zoroastrian

		# Has formed Eranshahr with a majority muslim population
		has_variable = fnr_zoroastrian_restored_muslim_reaction

		OR = {
			religious_unity < 0.25 # The realm is Zoroastrian by crown but not by population
			religion_percentage_in_country = { # There is a large non-Zoroastrian (Muslim) majority to revolt
				religion = religion:zoroastrian
				value < 0.25
			}
			OR = {
				stability < 10
				religious_unity < 0.35
			}
		}

		custom_tooltip = {
			text = had_zoroastrian_reaction_before_tt
			NOT = { has_variable = had_disaster_zoroastrian_reaction_variable }
		}
	}

	can_end = {
		fnr_zoroastrian_reaction_end_trigger = yes
	}

	modifier = {
		clergy_estate_target_satisfaction = huge_permanent_target_satisfaction_penalty
		monthly_war_exhaustion = 0.10
		tolerance_heretic = -2
	}

	on_start = {
		set_variable = {
			name = target_religious_unity_variable
			value = {
				add = religious_unity
				add = 0.2
				min = 0.55
				max = 0.85
			}
		}
		trigger_event_non_silently = flavor_ERSHR.1
	}

	on_monthly = {
		show_as_tooltip = {
			random_list = {
				94 = { custom_tooltip = no_event_occurs_tt }
				6 = { custom_tooltip = an_event_occurs_tt }
			}
		}
		hidden_effect = {
			random_list = {
				940 = { }
				10 = { trigger_event_silently = flavor_ERSHR.2 }	# Local uprising
				10 = { trigger_event_silently = flavor_ERSHR.3 }	# Clergy demand harsher measures
				10 = { trigger_event_silently = flavor_ERSHR.4 }	# Foreign Muslim power stirs unrest
				10 = { trigger_event_silently = flavor_ERSHR.5 }	# Moderate faction seeks tolerance
			}
		}
	}

	on_end = {
		set_variable = had_disaster_zoroastrian_reaction_variable
		remove_variable = fnr_zoroastrian_restored_muslim_reaction
		remove_variable = target_religious_unity_variable
		if = {
			limit = { religious_unity >= var:target_religious_unity_variable }
			trigger_event_non_silently = flavor_ERSHR.10	# Suppressed / converted: victory
		}
		else = {
			trigger_event_non_silently = flavor_ERSHR.11	# Forced to relent / tolerance
		}
	}
}

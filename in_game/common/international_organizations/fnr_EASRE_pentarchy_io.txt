# The Pentarchy - simple stabilizing IO for the Eastern Roman Empire
# Created by flavor_EASRE.2. ERE always leads. Orthodox countries may join.
# The leader's bonus scales with how many of the five patriarchal seats it owns
# (Rome, Constantinople, Alexandria, Antioch, Jerusalem).
# Councils, excommunication and Orthodox crusades from the content map are deliberately left out.

fnr_EASRE_pentarchy_io = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_type = country
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	# Leader: base stability package
	leader_modifier = {
		monthly_legitimacy = 0.1
		global_estate_target_satisfaction = small_permanent_target_satisfaction
		global_pop_conversion_speed_modifier = 0.05
	}

	# Leader: per patriarchal seat owned (0-5)
	leader_modifier = {
		scale = {
			value = 0
			if = { limit = { owns = location:rome } add = 1 }
			if = { limit = { owns = location:constantinople } add = 1 }
			if = { limit = { owns = location:alexandria } add = 1 }
			if = { limit = { owns = location:antioch } add = 1 }
			if = { limit = { owns = location:jerusalem } add = 1 }
		}
		monthly_prestige = 0.02
		monthly_legitimacy = 0.03
		global_pop_conversion_speed_modifier = 0.03
	}

	# Other members
	non_leader_modifier = {
		monthly_legitimacy = 0.05
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		tag = EASRE
	}

	can_lead_trigger = {
		tag = EASRE
	}

	can_join_trigger = {
		religion = religion:orthodox
		NOT = { is_member_of_international_organization = international_organization:fnr_EASRE_pentarchy_io }
	}

	can_leave_trigger = {
		NOT = { tag = EASRE }
	}

	auto_leave_trigger = {
		NOT = { religion = religion:orthodox }
		NOT = { tag = EASRE }
	}

	can_declare_war = {
		always = yes
	}

	ai_desire_to_join = {
		add = {
			desc = "BASE_DESIRE"
			value = -50
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 1000
			}
		}
		if = {
			limit = { exists = scope:actor }
			add = {
				desc = "DIPLOREASON_OPINION"
				value = {
					value = "opinion(scope:actor)"
					multiply = 0.5
				}
			}
		}
		if = {
			limit = {
				exists = scope:actor
				religion = scope:actor.religion
			}
			add = {
				desc = "DIPLOREASON_SAME_RELIGION"
				value = 40
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}

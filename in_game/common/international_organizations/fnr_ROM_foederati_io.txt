# The Foederati - Amici et Socii Populi Romani: Rome's client kings and frontier allies (Pax Romana)
# Founded by the fnr_ROM_found_foederati decision, or converted from the Republic's Socii when Rome forms
# (fnr_ROM_convert_socii_effect, from fnr_ROM.1). The Roman Empire always leads.
# Who can join: subjects of Rome (they join automatically), former Socii, and any country with its capital
# in a diocese region or the Caucasus. Members are called into each other's defensive wars.
# Actions in fnr_ROM_foederati_actions.txt. Client kingdoms (decisions) are created as vassals and join.

fnr_ROM_foederati_io = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_type = country
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	# Leader: base package
	leader_modifier = {
		global_manpower_modifier = 0.05
		global_defensive = 0.05
	}

	# Leader: per member, up to 10
	leader_modifier = {
		scale = {
			value = 0
			if = {
				limit = { has_variable = fnr_ROM_foederati_members }
				add = var:fnr_ROM_foederati_members
			}
			max = 10
		}
		global_manpower_modifier = 0.02
		monthly_prestige = 0.02
	}

	# Other members
	non_leader_modifier = {
		global_defensive = 0.10
		land_morale_modifier = 0.05
	}

	join_defensive_wars_can_call = {
		always = yes
	}

	join_defensive_wars_auto_call = {
		always = yes
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		has_or_had_tag = ROM
	}

	can_lead_trigger = {
		has_or_had_tag = ROM
	}

	can_join_trigger = {
		OR = {
			overlord ?= { has_or_had_tag = ROM }
			has_variable = fnr_ROM_former_socius
			capital ?= { fnr_ROM_in_diocese_region = yes }
			capital ?= { region = region:caucasus_region }
		}
		NOT = { is_member_of_international_organization = international_organization:fnr_ROM_foederati_io }
		NOT = {
			any_country_at_war_with = {
				is_member_of_international_organization = international_organization:fnr_ROM_foederati_io
			}
		}
	}

	can_leave_trigger = {
		NOT = { has_or_had_tag = ROM }
		NOT = { overlord ?= { has_or_had_tag = ROM } }
	}

	auto_leave_trigger = {
		NOT = { has_or_had_tag = ROM }
		NOT = { overlord ?= { has_or_had_tag = ROM } }
		NOT = { has_variable = fnr_ROM_former_socius }
		NOT = { capital ?= { fnr_ROM_in_diocese_region = yes } }
		NOT = { capital ?= { region = region:caucasus_region } }
	}

	# Rome's subjects join automatically
	monthly_effect = {
		leader_country ?= {
			every_subject = {
				limit = {
					NOT = { is_member_of_international_organization = international_organization:fnr_ROM_foederati_io }
				}
				save_scope_as = fnr_ROM_foederati_joiner
				international_organization:fnr_ROM_foederati_io = {
					add_country_to_international_organization = scope:fnr_ROM_foederati_joiner
				}
			}
		}
	}

	on_joined = {
		if = {
			limit = { NOT = { has_or_had_tag = ROM } }
			international_organization:fnr_ROM_foederati_io = {
				leader_country ?= { save_scope_as = fnr_ROM_foederati_leader }
			}
			if = {
				limit = { exists = scope:fnr_ROM_foederati_leader }
				add_opinion = { target = scope:fnr_ROM_foederati_leader modifier = fnr_ROM_sworn_foederatus }
			}
		}
	}

	can_declare_war = {
		always = yes
	}

	ai_desire_to_join = {
		add = {
			desc = "BASE_DESIRE"
			value = -30
		}
		if = {
			limit = { overlord ?= { has_or_had_tag = ROM } }
			add = {
				desc = "fnr_ROM_foederati_reason_subject"
				value = 100
			}
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 1000
			}
		}
		if = {
			limit = { exists = scope:actor }
			add = {
				desc = "DIPLOREASON_OPINION"
				value = {
					value = "opinion(scope:actor)"
					multiply = 0.5
				}
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}

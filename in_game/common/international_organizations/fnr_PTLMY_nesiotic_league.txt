# The League of the Islanders (Koinon ton Nesioton) - the Ptolemaic Kingdom's league of the Aegean,
# Cyprus, Cyrenaica and the Levantine coast (Pax Romana). Founded by fnr_PTLMY_found_nesiotic_league.
# The Ptolemaic Kingdom always leads. Its subjects join automatically. Actions in fnr_PTLMY_league_actions.txt.

fnr_PTLMY_nesiotic_league = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_type = country
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	leader_modifier = {
		navy_maintenance_efficiency = 0.05
		naval_morale_modifier = 0.05
	}

	leader_modifier = {
		scale = {
			value = 0
			if = {
				limit = { has_variable = fnr_PTLMY_league_members }
				add = var:fnr_PTLMY_league_members
			}
			max = 8
		}
		trade_income = 0.02
		monthly_prestige = 0.02
	}

	non_leader_modifier = {
		global_defensive = 0.10
		trade_income = 0.05
	}

	join_defensive_wars_can_call = {
		always = yes
	}

	join_defensive_wars_auto_call = {
		always = yes
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		has_or_had_tag = PTLMY
	}

	can_lead_trigger = {
		has_or_had_tag = PTLMY
	}

	can_join_trigger = {
		OR = {
			overlord ?= { has_or_had_tag = PTLMY }
			capital ?= { fnr_PTLMY_nesiotic_lands = yes }
		}
		NOT = { is_member_of_international_organization = international_organization:fnr_PTLMY_nesiotic_league }
		NOT = {
			any_country_at_war_with = {
				is_member_of_international_organization = international_organization:fnr_PTLMY_nesiotic_league
			}
		}
	}

	can_leave_trigger = {
		NOT = { has_or_had_tag = PTLMY }
		NOT = { overlord ?= { has_or_had_tag = PTLMY } }
	}

	auto_leave_trigger = {
		NOT = { has_or_had_tag = PTLMY }
		NOT = { overlord ?= { has_or_had_tag = PTLMY } }
		NOT = { capital ?= { fnr_PTLMY_nesiotic_lands = yes } }
	}

	monthly_effect = {
		leader_country ?= {
			every_subject = {
				limit = {
					NOT = { is_member_of_international_organization = international_organization:fnr_PTLMY_nesiotic_league }
				}
				save_scope_as = fnr_PTLMY_league_joiner
				international_organization:fnr_PTLMY_nesiotic_league = {
					add_country_to_international_organization = scope:fnr_PTLMY_league_joiner
				}
			}
		}
	}

	on_joined = {
		if = {
			limit = { NOT = { has_or_had_tag = PTLMY } }
			international_organization:fnr_PTLMY_nesiotic_league = {
				leader_country ?= { save_scope_as = fnr_PTLMY_league_leader }
			}
			if = {
				limit = { exists = scope:fnr_PTLMY_league_leader }
				add_opinion = { target = scope:fnr_PTLMY_league_leader modifier = fnr_PTLMY_sworn_islander }
			}
		}
	}

	can_declare_war = {
		always = yes
	}

	ai_desire_to_join = {
		add = {
			desc = "BASE_DESIRE"
			value = -30
		}
		if = {
			limit = { overlord ?= { has_or_had_tag = PTLMY } }
			add = {
				desc = "fnr_PTLMY_league_reason_subject"
				value = 100
			}
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 1000
			}
		}
		if = {
			limit = { exists = scope:actor }
			add = {
				desc = "DIPLOREASON_OPINION"
				value = {
					value = "opinion(scope:actor)"
					multiply = 0.5
				}
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}

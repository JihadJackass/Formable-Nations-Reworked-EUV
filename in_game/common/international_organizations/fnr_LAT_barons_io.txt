# The Barons of Romania - simple IO for the Latin Empire
# Founded by the fnr_LAT_found_barons_of_romania decision. LAT always leads.
# Catholic Frankokratia states and any subject of the Latin Empire may join; the Latin Empire's
# subjects (e.g. the Partition fiefs) join automatically. Members are called into each other's
# defensive wars. League actions: the Parliament of Ravennika and Summon the Barons decisions.

fnr_LAT_barons_io = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_type = country
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	# Leader: base package
	leader_modifier = {
		monthly_legitimacy = 0.05
		army_heavy_cavalry_power = 0.05
	}

	# Leader: per member (not counting the Latin Empire)
	leader_modifier = {
		scale = {
			value = 0
			if = {
				limit = { has_variable = fnr_LAT_barons_members }
				add = var:fnr_LAT_barons_members
			}
			max = 8
		}
		global_levy_size_modifier = 0.02
		monthly_prestige = 0.02
	}

	# Other members
	non_leader_modifier = {
		global_defensive = 0.10
		army_heavy_cavalry_power = 0.05
	}

	join_defensive_wars_can_call = {
		always = yes
	}

	join_defensive_wars_auto_call = {
		always = yes
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		tag = LAT
	}

	can_lead_trigger = {
		tag = LAT
	}

	can_join_trigger = {
		OR = {
			is_subject_of = c:LAT
			AND = {
				religion = religion:catholic
				is_frankokratia_state = yes
			}
		}
		NOT = { is_member_of_international_organization = international_organization:fnr_LAT_barons_io }
		NOT = {
			any_country_at_war_with = {
				is_member_of_international_organization = international_organization:fnr_LAT_barons_io
			}
		}
	}

	can_leave_trigger = {
		NOT = { tag = LAT }
	}

	auto_leave_trigger = {
		NOT = { tag = LAT }
		NOT = { is_subject_of = c:LAT }
		NOT = { religion = religion:catholic }
	}

	# The Latin Empire's subjects join automatically
	monthly_effect = {
		leader_country ?= {
			every_subject = {
				limit = {
					NOT = { is_member_of_international_organization = international_organization:fnr_LAT_barons_io }
				}
				save_scope_as = fnr_LAT_barons_joiner
				international_organization:fnr_LAT_barons_io = {
					add_country_to_international_organization = scope:fnr_LAT_barons_joiner
				}
			}
		}
	}

	on_joined = {
		if = {
			limit = { NOT = { tag = LAT } }
			add_opinion = { target = c:LAT modifier = fnr_LAT_sworn_baron }
		}
	}

	can_declare_war = {
		always = yes
	}

	ai_desire_to_join = {
		add = {
			desc = "BASE_DESIRE"
			value = -30
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 1000
			}
		}
		if = {
			limit = { exists = scope:actor }
			add = {
				desc = "DIPLOREASON_OPINION"
				value = {
					value = "opinion(scope:actor)"
					multiply = 0.5
				}
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}

# Concilium Occidentis - simple defensive IO for the Western Roman Empire
# Created by the fnr_WESRE_convene_concilium decision. WRE always leads.
# Christian countries that border the WRE or a member, and are not at war with a member, may join.
# Members are called into each other's defensive wars.
# The leader's bonus scales with the member count (variable kept by the yearly pulse).
# Council sessions and the Defense/Trade/Faith votes from the content map are deliberately left out.
# Not adapted: the 8-member cap and Foederati States as a separate membership tier.

fnr_WESRE_concilium_io = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_type = country
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	# Leader: base package
	leader_modifier = {
		global_defensive = 0.05
		diplomatic_reputation = diplomatic_reputation_weak_bonus
	}

	# Leader: per member (not counting the WRE)
	leader_modifier = {
		scale = {
			value = 0
			if = {
				limit = { has_variable = fnr_WESRE_concilium_members }
				add = var:fnr_WESRE_concilium_members
			}
			max = 8
		}
		monthly_legitimacy = 0.02
		monthly_prestige = 0.02
	}

	# Other members
	non_leader_modifier = {
		global_defensive = 0.10
		antagonism_received_modifier = -0.10
	}

	join_defensive_wars_can_call = {
		always = yes
	}

	join_defensive_wars_auto_call = {
		always = yes
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		tag = WESRE
	}

	can_lead_trigger = {
		tag = WESRE
	}

	can_join_trigger = {
		religion.group = religion_group:christian
		NOT = { is_member_of_international_organization = international_organization:fnr_WESRE_concilium_io }
		OR = {
			is_neighbor_of = c:WESRE
			any_neighbor_country = {
				is_member_of_international_organization = international_organization:fnr_WESRE_concilium_io
			}
		}
		NOT = {
			any_country_at_war_with = {
				is_member_of_international_organization = international_organization:fnr_WESRE_concilium_io
			}
		}
	}

	can_leave_trigger = {
		NOT = { tag = WESRE }
	}

	auto_leave_trigger = {
		NOT = { religion.group = religion_group:christian }
		NOT = { tag = WESRE }
	}

	can_declare_war = {
		always = yes
	}

	ai_desire_to_join = {
		add = {
			desc = "BASE_DESIRE"
			value = -40
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 1000
			}
		}
		if = {
			limit = { exists = scope:actor }
			add = {
				desc = "DIPLOREASON_OPINION"
				value = {
					value = "opinion(scope:actor)"
					multiply = 0.5
				}
			}
		}
		if = {
			limit = {
				exists = scope:actor
				religion = scope:actor.religion
			}
			add = {
				desc = "DIPLOREASON_SAME_RELIGION"
				value = 30
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}

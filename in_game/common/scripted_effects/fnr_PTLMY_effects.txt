# Ptolemaic Kingdom scripted effects (Pax Romana)

# ---------- Basileus and Pharaoh (fnr_PTLMY_crown, -100 Pharaoh to +100 Basileus) ----------

fnr_PTLMY_clear_crown_tiers_effect = {
	if = { limit = { has_country_modifier = fnr_PTLMY_crown_basileus } remove_country_modifier = fnr_PTLMY_crown_basileus }
	if = { limit = { has_country_modifier = fnr_PTLMY_crown_greek } remove_country_modifier = fnr_PTLMY_crown_greek }
	if = { limit = { has_country_modifier = fnr_PTLMY_crown_dual } remove_country_modifier = fnr_PTLMY_crown_dual }
	if = { limit = { has_country_modifier = fnr_PTLMY_crown_egyptian } remove_country_modifier = fnr_PTLMY_crown_egyptian }
	if = { limit = { has_country_modifier = fnr_PTLMY_crown_pharaoh } remove_country_modifier = fnr_PTLMY_crown_pharaoh }
}

fnr_PTLMY_refresh_crown_tier_effect = {
	hidden_effect = {
		fnr_PTLMY_clear_crown_tiers_effect = yes
		if = {
			limit = { has_variable = fnr_PTLMY_crown }
			if = {
				limit = { var:fnr_PTLMY_crown >= 60 }
				add_country_modifier = { modifier = fnr_PTLMY_crown_basileus years = -1 }
			}
			else_if = {
				limit = { var:fnr_PTLMY_crown >= 20 }
				add_country_modifier = { modifier = fnr_PTLMY_crown_greek years = -1 }
			}
			else_if = {
				limit = { var:fnr_PTLMY_crown >= -19 }
				add_country_modifier = { modifier = fnr_PTLMY_crown_dual years = -1 }
			}
			else_if = {
				limit = { var:fnr_PTLMY_crown >= -59 }
				add_country_modifier = { modifier = fnr_PTLMY_crown_egyptian years = -1 }
			}
			else = {
				add_country_modifier = { modifier = fnr_PTLMY_crown_pharaoh years = -1 }
			}
		}
	}
}

# Silent change, kept in range. Use with VALUE
fnr_PTLMY_shift_crown_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = fnr_PTLMY_crown } }
			set_variable = { name = fnr_PTLMY_crown value = 0 }
		}
		change_variable = { name = fnr_PTLMY_crown add = $VALUE$ }
		if = {
			limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown > 100 } trigger_else = { always = no } }
			set_variable = { name = fnr_PTLMY_crown value = 100 }
		}
		if = {
			limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown < -100 } trigger_else = { always = no } }
			set_variable = { name = fnr_PTLMY_crown value = -100 }
		}
	}
}

# Visible change with one tooltip line (fnr_PTLMY_crown_<TT>_tt). Use with VALUE and TT
fnr_PTLMY_change_crown_effect = {
	custom_tooltip = fnr_PTLMY_crown_$TT$_tt
	fnr_PTLMY_shift_crown_effect = { VALUE = $VALUE$ }
	fnr_PTLMY_refresh_crown_tier_effect = yes
}

# ---------- The Library of Alexandria (fnr_PTLMY_scholarship, 0 and up) ----------

fnr_PTLMY_clear_scholarship_tiers_effect = {
	if = { limit = { has_country_modifier = fnr_PTLMY_library_tier_4 } remove_country_modifier = fnr_PTLMY_library_tier_4 }
	if = { limit = { has_country_modifier = fnr_PTLMY_library_tier_3 } remove_country_modifier = fnr_PTLMY_library_tier_3 }
	if = { limit = { has_country_modifier = fnr_PTLMY_library_tier_2 } remove_country_modifier = fnr_PTLMY_library_tier_2 }

}

fnr_PTLMY_refresh_scholarship_tier_effect = {
	hidden_effect = {
		fnr_PTLMY_clear_scholarship_tiers_effect = yes
		if = {
			limit = { has_variable = fnr_PTLMY_scholarship }
			if = {
				limit = { var:fnr_PTLMY_scholarship >= 150 }
				add_country_modifier = { modifier = fnr_PTLMY_library_tier_4 years = -1 }
			}
			else_if = {
				limit = { var:fnr_PTLMY_scholarship >= 75 }
				add_country_modifier = { modifier = fnr_PTLMY_library_tier_3 years = -1 }
			}
			else_if = {
				limit = { var:fnr_PTLMY_scholarship >= 25 }
				add_country_modifier = { modifier = fnr_PTLMY_library_tier_2 years = -1 }
			}
			else = {
				add_country_modifier = { modifier = fnr_PTLMY_library_tier_1 years = -1 }
			}
		}
	}
}

# Called from the PTLMY_f form_effect (once). Origin, starting values, opening event.
fnr_PTLMY_formation_effect = {
	if = {
		limit = { NOT = { has_variable = fnr_PTLMY_formed } }
		hidden_effect = {
			set_variable = fnr_PTLMY_formed
			if = {
				limit = { has_variable = fnr_PTLMY_origin_bactrian }
				set_variable = { name = fnr_PTLMY_crown value = 50 }
				set_variable = { name = fnr_PTLMY_scholarship value = 15 }
			}
			else_if = {
				limit = { has_variable = fnr_PTLMY_hellenized }
				set_variable = fnr_PTLMY_origin_egyptian
				set_variable = { name = fnr_PTLMY_crown value = -30 }
				set_variable = { name = fnr_PTLMY_scholarship value = 5 }
			}
			else = {
				set_variable = fnr_PTLMY_origin_greek
				set_variable = { name = fnr_PTLMY_crown value = 30 }
				set_variable = { name = fnr_PTLMY_scholarship value = 10 }
			}
			set_variable = { name = fnr_PTLMY_granary value = 1 }
		}
		fnr_PTLMY_refresh_crown_tier_effect = yes
		fnr_PTLMY_refresh_scholarship_tier_effect = yes
		custom_tooltip = fnr_PTLMY_formation_tt
		trigger_event_non_silently = fnr_PTLMY.1
	}
}

# ---------- Yearly upkeep ----------

fnr_PTLMY_yearly_crown_effect = {
	if = {
		limit = { NOT = { has_variable = fnr_PTLMY_crown } }
		set_variable = { name = fnr_PTLMY_crown value = 0 }
	}
	# Drift 2 a year towards the Double Crown (1 when only 1 away)
	if = {
		limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown > 1 } trigger_else = { always = no } }
		fnr_PTLMY_shift_crown_effect = { VALUE = -2 }
	}
	else_if = {
		limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown > 0 } trigger_else = { always = no } }
		fnr_PTLMY_shift_crown_effect = { VALUE = -1 }
	}
	else_if = {
		limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown < -1 } trigger_else = { always = no } }
		fnr_PTLMY_shift_crown_effect = { VALUE = 2 }
	}
	else_if = {
		limit = { trigger_if = { limit = { has_variable = fnr_PTLMY_crown } var:fnr_PTLMY_crown < 0 } trigger_else = { always = no } }
		fnr_PTLMY_shift_crown_effect = { VALUE = 1 }
	}
	# The seat of the court
	if = {
		limit = { capital = location:alexandria }
		fnr_PTLMY_shift_crown_effect = { VALUE = 1 }
	}
	else = {
		fnr_PTLMY_shift_crown_effect = { VALUE = -1 }
	}
	# Privileges
	if = {
		limit = { has_estate_privilege = estate_privilege:fnr_PTLMY_kleruchoi }
		fnr_PTLMY_shift_crown_effect = { VALUE = 2 }
	}
	if = {
		limit = { has_estate_privilege = estate_privilege:fnr_PTLMY_priests_of_ptah }
		fnr_PTLMY_shift_crown_effect = { VALUE = -2 }
	}
	if = {
		limit = { has_estate_privilege = estate_privilege:fnr_PTLMY_machimoi }
		fnr_PTLMY_shift_crown_effect = { VALUE = -1 }
	}
	# Reforms
	if = {
		limit = { has_reform = government_reform:fnr_PTLMY_hellenistic_basileia }
		fnr_PTLMY_shift_crown_effect = { VALUE = 2 }
	}
	if = {
		limit = { has_reform = government_reform:fnr_PTLMY_pharaonic_kingship }
		fnr_PTLMY_shift_crown_effect = { VALUE = -2 }
	}
	# The Royal Economy
	if = {
		limit = { has_policy = fnr_PTLMY_greek_tax_farmers_policy }
		fnr_PTLMY_shift_crown_effect = { VALUE = 1 }
	}
	else_if = {
		limit = { has_policy = fnr_PTLMY_temple_economy_policy }
		fnr_PTLMY_shift_crown_effect = { VALUE = -1 }
	}
	fnr_PTLMY_refresh_crown_tier_effect = yes
}

# One outcome. Use with OUTCOME = bountiful / good / low / failed
fnr_PTLMY_nile_outcome_effect = {
	if = {
		limit = { always = $IS_FAILED$ }
		if = {
			limit = { NOT = { fnr_PTLMY_granary_below = { VALUE = 1 } } }
			change_variable = { name = fnr_PTLMY_granary add = -1 }
			add_country_modifier = { modifier = fnr_PTLMY_nile_granaries_opened years = 1 }
			trigger_event_non_silently = fnr_PTLMY.32
		}
		else = {
			add_country_modifier = { modifier = fnr_PTLMY_nile_failed years = 1 }
			trigger_event_non_silently = fnr_PTLMY.31
		}
	}
	else = {
		add_country_modifier = { modifier = fnr_PTLMY_nile_$OUTCOME$ years = 1 }
	}
}

# Yearly: the Nile flood. Nilometers and the Canal of the Pharaohs make good years likelier.
fnr_PTLMY_yearly_nile_effect = {
	if = {
		limit = { region:egypt_region = { any_location_in_region = { owner = root } } }
		fnr_PTLMY_clear_nile_effect = yes
		if = {
			limit = {
				any_owned_location = {
					count >= 2
					has_building = building_type:fnr_PTLMY_nilometer
				}
				has_variable = fnr_PTLMY_canal
			}
			random_list = {
				30 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = bountiful IS_FAILED = no } trigger_event_non_silently = fnr_PTLMY.30 }
				50 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = good IS_FAILED = no } }
				17 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = low IS_FAILED = no } }
				3 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = failed IS_FAILED = yes } }
			}
		}
		else_if = {
			limit = {
				OR = {
					any_owned_location = { has_building = building_type:fnr_PTLMY_nilometer }
					has_variable = fnr_PTLMY_canal
				}
			}
			random_list = {
				25 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = bountiful IS_FAILED = no } trigger_event_non_silently = fnr_PTLMY.30 }
				48 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = good IS_FAILED = no } }
				21 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = low IS_FAILED = no } }
				6 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = failed IS_FAILED = yes } }
			}
		}
		else = {
			random_list = {
				20 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = bountiful IS_FAILED = no } trigger_event_non_silently = fnr_PTLMY.30 }
				45 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = good IS_FAILED = no } }
				25 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = low IS_FAILED = no } }
				10 = { fnr_PTLMY_nile_outcome_effect = { OUTCOME = failed IS_FAILED = yes } }
			}
		}
	}
}

# Yearly: League of the Islanders member count (not counting us)
fnr_PTLMY_count_league_effect = {
	set_variable = { name = fnr_PTLMY_league_members value = 0 }
	if = {
		limit = { exists = international_organization:fnr_PTLMY_nesiotic_league }
		every_country = {
			limit = {
				NOT = { this = root }
				is_member_of_international_organization = international_organization:fnr_PTLMY_nesiotic_league
			}
			root = { change_variable = { name = fnr_PTLMY_league_members add = 1 } }
		}
	}
}

# ---------- Rebellions ----------

fnr_PTLMY_province_joins_rebels_effect = {
	if = {
		limit = {
			exists = scope:$LOCATION$
			exists = scope:fnr_PTLMY_rebel
		}
		scope:$LOCATION$.province = {
			every_location_in_province = {
				limit = { owner = root }
				every_pop = {
					limit = { owner = root }
					add_pop_satisfaction = pop_satisfaction_ultimate_penalty
					change_pop_allegiance = scope:fnr_PTLMY_rebel
				}
			}
		}
	}
}

# A pretender rebellion in up to four provinces of one area (falls back to any non-capital land).
# Use with NAME = rebel name, AREA = area key, FEMALE_OK = yes/no (a Lagid princess may lead)
fnr_PTLMY_rebellion_effect = {
	hidden_effect = {
		random_owned_location = {
			limit = {
				is_capital = no
				area = area:$AREA$
			}
			save_scope_as = fnr_PTLMY_revolt_1
		}
		if = {
			limit = { NOT = { exists = scope:fnr_PTLMY_revolt_1 } }
			random_owned_location = {
				limit = { is_capital = no }
				save_scope_as = fnr_PTLMY_revolt_1
			}
		}
	}
	if = {
		limit = { exists = scope:fnr_PTLMY_revolt_1 }
		create_rebel = {
			category = pretender
			name = $NAME$
			save_scope_as = fnr_PTLMY_rebel
		}
	}
	if = {
		limit = { exists = scope:fnr_PTLMY_rebel }
		hidden_effect = {
			random_character = {
				limit = {
					is_alive = yes
					age_in_years >= 16
					NOT = { this = root.ruler }
					trigger_if = {
						limit = { always = $FEMALE_OK$ }
						always = yes
					}
					trigger_else = { is_female = no }
				}
				change_character_allegiance = scope:fnr_PTLMY_rebel
			}
			random_owned_location = {
				limit = {
					is_capital = no
					area = scope:fnr_PTLMY_revolt_1.area
					NOT = { this.province = scope:fnr_PTLMY_revolt_1.province }
				}
				save_scope_as = fnr_PTLMY_revolt_2
			}
			random_owned_location = {
				limit = {
					is_capital = no
					area = scope:fnr_PTLMY_revolt_1.area
					NOT = { this.province = scope:fnr_PTLMY_revolt_1.province }
					NOT = { scope:fnr_PTLMY_revolt_2 ?= { province = prev.province } }
				}
				save_scope_as = fnr_PTLMY_revolt_3
			}
			random_owned_location = {
				limit = {
					is_capital = no
					NOT = { this.province = scope:fnr_PTLMY_revolt_1.province }
					NOT = { scope:fnr_PTLMY_revolt_2 ?= { province = prev.province } }
					NOT = { scope:fnr_PTLMY_revolt_3 ?= { province = prev.province } }
					area = scope:fnr_PTLMY_revolt_1.area
				}
				save_scope_as = fnr_PTLMY_revolt_4
			}
			fnr_PTLMY_province_joins_rebels_effect = { LOCATION = fnr_PTLMY_revolt_1 }
			fnr_PTLMY_province_joins_rebels_effect = { LOCATION = fnr_PTLMY_revolt_2 }
			fnr_PTLMY_province_joins_rebels_effect = { LOCATION = fnr_PTLMY_revolt_3 }
			fnr_PTLMY_province_joins_rebels_effect = { LOCATION = fnr_PTLMY_revolt_4 }
			scope:fnr_PTLMY_rebel ?= { add_rebel_progress = 1 }
			add_country_modifier = { modifier = fnr_rising_imminent years = 1 mode = add_and_extend }
		}
	}
}

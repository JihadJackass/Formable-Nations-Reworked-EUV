# Ptolemaic Kingdom Building Types (Pax Romana)

# The Mouseion - Alexandria only. Scholarship +2 a year.
fnr_PTLMY_mouseion = {
	audio_tier = 3
	is_foreign = no
	max_levels = 1
	pop_type = burghers
	forbidden_for_estates = yes

	employment_size = large_noble_employment
	category = government_category

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	country_potential = {
		has_or_had_tag = PTLMY
		has_advance = PTLMY_mouseion
	}

	allow = {
		this = location:alexandria
	}

	unique_production_methods = {
		fnr_PTLMY_mouseion_maintenance = {
			books = 0.05
			marble = 0.03
			category = building_maintenance
		}
	}

	construction_demand = early_capital_building_construction

	modifier = {
		local_max_control = 0.05
		local_monthly_prosperity = 0.05
	}
}

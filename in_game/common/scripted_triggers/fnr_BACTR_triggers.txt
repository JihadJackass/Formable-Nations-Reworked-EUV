# Enough land to carve the client KEY from (COUNT non-capital locations), and our capital is not in it
fnr_BACTR_can_carve_client = {
	custom_tooltip = {
		text = fnr_BACTR_client_enough_land_$COUNT$_tt
		any_owned_location = {
			count >= $COUNT$
			is_capital = no
			fnr_BACTR_client_area_$KEY$ = yes
		}
	}
	custom_tooltip = {
		text = fnr_BACTR_client_not_capital_tt
		capital = { fnr_BACTR_client_area_$KEY$ = no }
	}
}

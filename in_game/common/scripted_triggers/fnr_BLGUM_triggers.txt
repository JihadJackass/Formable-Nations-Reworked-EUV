# Country scope: a foreign crown holds a large part of the Belgian lands
fnr_BLGUM_rules_the_south = {
	NOT = { tag = BLGUM }
	any_owned_location = {
		fnr_BLGUM_is_belgian_location = yes
		count >= 12
	}
	capital ?= { fnr_BLGUM_is_belgian_location = no }
}

# Independence at least / below VALUE (false before the disaster starts)
fnr_BLGUM_independence_at_least = {
	trigger_if = {
		limit = { has_variable = fnr_BLGUM_independence }
		var:fnr_BLGUM_independence >= $VALUE$
	}
	trigger_else = { always = no }
}
fnr_BLGUM_independence_below = {
	trigger_if = {
		limit = { has_variable = fnr_BLGUM_independence }
		var:fnr_BLGUM_independence < $VALUE$
	}
	trigger_else = { always = no }
}

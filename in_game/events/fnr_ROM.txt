namespace = fnr_ROM

# ==========================================================================
# Pax Romana - Roman Empire: "Imperium Romanum"
#  .1  Ave Imperator                - on formation (fnr_ROM_formation_effect): the origin of this Rome
#  .2  The Legions Proclaim an Emperor - Legion Loyalty below 40 (yearly pulses): a usurper
#  .3  The Praetorians Stir         - Praetorian Guard + Legion Loyalty below 50 (yearly pulse)
#  .4  The Usurper Falls            - our usurper rebels are gone (yearly pulse)
#  .5  Foederati at the Limes       - frontier flavour (yearly pulse)
#  .6  The Census of the Empire     - frontier flavour (yearly pulse, 20-year cooldown)
#  .7  The Crisis of the Third Century - start of the disaster
#  .8  The Debasement of the Coinage   - disaster monthly
#  .9  The Gallic Empire            - disaster monthly, once: the West secedes
#  .10 The Palmyrene Empire         - disaster monthly, once: the East secedes
#  .11 A Soldier Emperor            - disaster monthly
#  .12 The Barbarians Cross the Frontier - disaster monthly
#  .13 Restitutor Orbis             - end of the disaster
#  .14 Imperium Sine Fine           - all eleven dioceses restored
#  .15 The Senate of Rome           - frontier flavour (yearly pulse, once)
# ==========================================================================

# 1 - Ave Imperator
fnr_ROM.1 = {
	type = country_event
	title = fnr_ROM.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = fnr_ROM_origin_great_republic }
				desc = fnr_ROM.1.desc.great_republic
			}
			triggered_desc = {
				trigger = { has_variable = fnr_ROM_origin_augustus }
				desc = fnr_ROM.1.desc.augustus
			}
			triggered_desc = {
				trigger = { has_variable = fnr_ROM_origin_principate }
				desc = fnr_ROM.1.desc.principate
			}
			triggered_desc = {
				trigger = { has_variable = fnr_ROM_origin_renovatio }
				desc = fnr_ROM.1.desc.renovatio
			}
			triggered_desc = {
				trigger = { has_variable = fnr_ROM_origin_new_rome }
				desc = fnr_ROM.1.desc.new_rome
			}
			triggered_desc = {
				desc = fnr_ROM.1.desc.restorer
			}
		}
	}
	outcome = positive

	trigger = { # fired from the form_effect, before the tag has changed
		NOT = { has_variable = fnr_seen_fnr_ROM_1 } # FNR: once per country (was fire_only_once)
		has_variable = fnr_ROM_formed
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		hidden_effect = { set_variable = fnr_seen_fnr_ROM_1 }
		# The Great Republic's Senate reform (added here as well: the formation effect runs before the tag changes)
		if = {
			limit = {
				has_variable = fnr_ROM_origin_great_republic
				NOT = { has_reform = government_reform:fnr_ROM_senate_of_generals }
			}
			add_reform = government_reform:fnr_ROM_senate_of_generals
		}
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
		if = {
			limit = { has_variable = fnr_ROM_origin_great_republic }
			add_country_modifier = { modifier = fnr_ROM_origin_great_republic years = -1 }
		}
		else_if = {
			limit = { has_variable = fnr_ROM_origin_augustus }
			add_country_modifier = { modifier = fnr_ROM_origin_augustus years = -1 }
		}
		else_if = {
			limit = { has_variable = fnr_ROM_origin_principate }
			add_country_modifier = { modifier = fnr_ROM_origin_principate years = -1 }
		}
		else_if = {
			limit = { has_variable = fnr_ROM_origin_renovatio }
			add_country_modifier = { modifier = fnr_ROM_origin_renovatio years = -1 }
		}
		else_if = {
			limit = { has_variable = fnr_ROM_origin_new_rome }
			add_country_modifier = { modifier = fnr_ROM_origin_new_rome years = -1 }
		}
		else = {
			add_country_modifier = { modifier = fnr_ROM_origin_restorer years = -1 }
		}
		fnr_ROM_refresh_legion_tier_effect = yes
		fnr_ROM_convert_socii_effect = yes
	}

	# A - To the frontiers
	option = {
		name = fnr_ROM.1.a
		trigger = { NOT = { has_variable = fnr_ROM_origin_great_republic } }
		ai_chance = { factor = 50 }
		fnr_ROM_change_loyalty_effect = { VALUE = 10 TT = plus_10 }
		add_army_tradition = army_tradition_mild_bonus
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# B - To the Forum
	option = {
		name = fnr_ROM.1.b
		trigger = { NOT = { has_variable = fnr_ROM_origin_great_republic } }
		ai_chance = { factor = 50 }
		add_legitimacy = legitimacy_mild_bonus
		add_stability = stability_weak_bonus
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# C - Great Republic name
	option = {
		name = fnr_ROM.1.c
		trigger = { has_variable = fnr_ROM_origin_great_republic }
		ai_chance = { factor = 25 }
		fnr_ROM_great_republic_name_effect = { NAME = fnr_ROM_GREAT_ROMAN_REPUBLIC_NAME }
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# D - Great Republic name
	option = {
		name = fnr_ROM.1.d
		trigger = { has_variable = fnr_ROM_origin_great_republic }
		ai_chance = { factor = 25 }
		fnr_ROM_great_republic_name_effect = { NAME = fnr_ROM_SERENE_REPUBLIC_NAME }
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# E - Great Republic name
	option = {
		name = fnr_ROM.1.e
		trigger = { has_variable = fnr_ROM_origin_great_republic }
		ai_chance = { factor = 25 }
		fnr_ROM_great_republic_name_effect = { NAME = fnr_ROM_IMPERIAL_REPUBLIC_NAME }
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# F - Great Republic name
	option = {
		name = fnr_ROM.1.f
		trigger = { has_variable = fnr_ROM_origin_great_republic }
		ai_chance = { factor = 25 }
		fnr_ROM_great_republic_name_effect = { NAME = fnr_ROM_REPUBLIC_OF_THE_ROMANS_NAME }
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# G - Great Republic name
	option = {
		name = fnr_ROM.1.g
		trigger = { has_variable = fnr_ROM_origin_great_republic }
		ai_chance = { factor = 25 }
		fnr_ROM_great_republic_name_effect = { NAME = fnr_ROM_GRAND_ROMAN_REPUBLIC_NAME }
		custom_tooltip = fnr_ROM_empire_systems_tt
	}

	# H - Keep the name we already have
	option = {

# 2 - The Legions Proclaim an Emperor
fnr_ROM.2 = {
	type = country_event
	title = fnr_ROM.2.title
	desc = fnr_ROM.2.desc
	outcome = negative

	trigger = {
		has_or_had_tag = ROM
		NOT = { has_variable = fnr_ROM_usurper_active }
		NOT = { has_variable = fnr_ROM_usurper_cooldown }
		in_civil_war = no
		any_owned_location = { is_capital = no }
	}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		set_variable = { name = fnr_ROM_usurper_cooldown years = 10 }
	}

	# A - Crush the usurper
	option = {
		name = fnr_ROM.2.a
		ai_chance = { factor = 50 }
		add_legitimacy = legitimacy_weak_penalty
		hidden_effect = { fnr_ROM_pick_far_location_effect = yes }
		fnr_ROM_raise_usurper_effect = { NAME = fnr_ROM_usurper_rebels }
		custom_tooltip = fnr_ROM_usurper_tt
	}

	# B - Buy back his legions
	option = {
		name = fnr_ROM.2.b
		trigger = { gold >= 300 }
		ai_chance = { factor = 50 }
		add_gold = -300
		add_prestige = prestige_mild_penalty
		fnr_ROM_change_loyalty_effect = { VALUE = 15 TT = plus_15 }
	}

	# C - Name him Caesar (Tetrarchy)
	option = {
		name = fnr_ROM.2.c
		trigger = { fnr_ROM_has_tetrarchy = yes }
		ai_chance = { factor = 100 }

# 4 - The Usurper Falls
fnr_ROM.4 = {
	type = country_event
	title = fnr_ROM.4.title
	desc = fnr_ROM.4.desc
	outcome = positive

	trigger = {
		has_or_had_tag = ROM
		has_variable = fnr_ROM_usurper_active
		in_civil_war = no
		custom_tooltip = {
			text = fnr_ROM_usurpers_gone_tt
			fnr_ROM_usurpers_in_field = no
		}
		# He did not win: the usurper is not on our throne
		trigger_if = {
			limit = { has_variable = fnr_ROM_usurper_leader }
			ruler ?= { NOT = { this = root.var:fnr_ROM_usurper_leader } }
		}
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
		remove_variable = fnr_ROM_usurper_active
		if = {
			limit = { has_variable = fnr_ROM_usurper_leader }
			remove_variable = fnr_ROM_usurper_leader
		}
	}

	# A - Damnatio memoriae
	option = {
		name = fnr_ROM.4.a
		ai_chance = { factor = 50 }

namespace = flavor_BACTR

# Event for formation of Greco-Buddhism
flavor_BACTR.1 = {
	type = country_event
	trigger = {
		NOT = { has_variable = fnr_seen_flavor_BACTR_1 } # FNR: once per country (was fire_only_once)
	}
	title = flavor_BACTR.1.title
	desc = flavor_BACTR.1.desc
	outcome = neutral
	immediate = {
		hidden_effect = { set_variable = fnr_seen_flavor_BACTR_1 }
	}
	option = {
		name = flavor_BACTR.1.a
		change_religion = religion:fnr_greco_buddhism_religion
		custom_tooltip = { text = BACTR_formable_trigger_1_tt }
		hidden_effect = { # Conversion to Zoroastrain
			every_pop = {
				limit = {
					religion = religion:sunni
					#religion.group = religion_group:muslim
				}
				split_pop = {
					fraction = 0.20
					religion = religion:zoroastrian
				}
			}
		}
		custom_tooltip = { text = BACTR_formable_trigger_2_tt }
		hidden_effect = { # Conversion to Greco-Buddhism
			every_pop = {
				limit = {
					religion = religion:sunni
					#religion.group = religion_group:muslim
				}
				split_pop = {
					fraction = 0.20
					religion = religion:fnr_greco_buddhism_religion
				}
			}
		}
	}
}

# Bactria Synod IO event from generic action
flavor_BACTR.3 = {
	type = country_event
	title = flavor_BACTR.3.title
	desc = flavor_BACTR.3.desc
	fire_only_once = yes
	outcome = neutral
	option = { 
		name = flavor_BACTR.3.a
		custom_tooltip = { text = BACTR_formable_trigger_3_tt }
		hidden_effect = {
			create_international_organization = {
				type = international_organization_type:fnr_bactria_synod_io
				add_country_to_international_organization = root
			}
			international_organization:fnr_bactria_synod_io = {
				set_leader_country = root
			}
		}
	}
}

# Honoring Alexander - the Macedonian Empire (decision fnr_BACTR_heir_of_alexander)
flavor_BACTR.9 = {
	type = country_event
	title = flavor_BACTR.9.title
	desc = flavor_BACTR.9.desc
	outcome = positive

	option = {
		name = flavor_BACTR.9.a
		hidden_effect = {
			remove_variable = fnr_BACTR_heir_pending
			set_variable = fnr_BACTR_heir_of_alexander
		}
		change_country_name = MCDEM
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
	}
}

# ==========================================================================
# Pax Romana - Dacia: "Heirs of Burebista"
#  .1  Heirs of Burebista       - on formation (DACIA_f form_effect)
#  .2  The Soul of Dacia        - 30 days later: Zalmoxian Revival or Daco-Roman Heritage
#  .3  The Gold of the Carpathians - 2-4 years after .2
#  .4  The Mountain of Zalmoxis - 5-8 years after .2, Zalmoxian Revival
#  .5  The Column of Trajan     - 5-8 years after .2, Daco-Roman Heritage
# ==========================================================================

# 1 - Heirs of Burebista
fnr_DACIA.1 = {
	type = country_event
	trigger = {
		NOT = { has_variable = fnr_seen_fnr_DACIA_1 } # FNR: once per country (was fire_only_once)
	}
	title = fnr_DACIA.1.title
	desc = fnr_DACIA.1.desc
	outcome = positive

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		hidden_effect = { set_variable = fnr_seen_fnr_DACIA_1 }
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		hidden_effect = {
			trigger_event_non_silently = { id = fnr_DACIA.2 days = 30 }
		}
	}

	option = {
		name = fnr_DACIA.1.a
		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = fnr_DACIA_burebistas_heir years = 15 mode = replace }
	}
}

# 2 - The Soul of Dacia (path choice)
fnr_DACIA.2 = {
	type = country_event
	title = fnr_DACIA.2.title
	desc = fnr_DACIA.2.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = DACIA
		NOT = { has_variable = fnr_DACIA_path_zalmoxian }
		NOT = { has_variable = fnr_DACIA_path_dacoroman }
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }
	}

	# A - Zalmoxian Revival. Christian realms pay in stability and clergy anger; pagan ones gain legitimacy.
	option = {
		name = fnr_DACIA.2.a
		ai_chance = { factor = 40 }
		set_variable = fnr_DACIA_path_zalmoxian
		if = {
			limit = {
				OR = {
					religion.group = religion_group:christian
					religion.group = religion_group:muslim
				}
			}
			add_stability = stability_mild_penalty
			add_estate_satisfaction = {
				type = estate_type:clergy_estate
				value = estate_satisfaction_mild_penalty
			}
		}
		else = {
			add_legitimacy = legitimacy_mild_bonus
		}
		change_religion = religion:fnr_zalmoxian_religion
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:fnr_zalmoxian_religion }
		add_country_modifier = { modifier = fnr_DACIA_zalmoxian_revival years = 20 mode = replace }
		custom_tooltip = fnr_DACIA_path_zalmoxian_tt
		hidden_effect = {
			every_pop = {
				limit = {
					owner ?= root
					NOT = { religion = religion:fnr_zalmoxian_religion }
				}
				if = {
					limit = {
						OR = {
							culture = culture:wallachian
							culture = culture:moldovan
							culture = culture:transylvanian
							culture = culture:istroromanian
						}
					}
					split_pop = {
						fraction = 0.25
						religion = religion:fnr_zalmoxian_religion
					}
				}
				else = {
					split_pop = {
						fraction = 0.10
						religion = religion:fnr_zalmoxian_religion
					}
				}
			}
			trigger_event_non_silently = { id = fnr_DACIA.3 days = { 730 1460 } }
			trigger_event_non_silently = { id = fnr_DACIA.4 days = { 1825 2920 } }
		}
	}

	# B - Daco-Roman Heritage. Keeps the faith; the Latin roots of the Romanian peoples.
	option = {
		name = fnr_DACIA.2.b
		ai_chance = { factor = 60 }
		set_variable = fnr_DACIA_path_dacoroman
		add_legitimacy = legitimacy_mild_bonus
		add_country_modifier = { modifier = fnr_DACIA_dacoroman_heritage years = 20 mode = replace }
		custom_tooltip = fnr_DACIA_path_dacoroman_tt
		hidden_effect = {
			trigger_event_non_silently = { id = fnr_DACIA.3 days = { 730 1460 } }
			trigger_event_non_silently = { id = fnr_DACIA.5 days = { 1825 2920 } }
		}
	}
}

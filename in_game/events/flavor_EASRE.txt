# ==========================================================================
# Pax Romana - Eastern Roman Empire: "Justinian's Shadow"
#  .1  The Purple Chamber             - on formation (EASRE_f form_effect)
#  .2  The Patriarch's Question       - 1 year later, picks Path A/B, founds the Pentarchy
#  .3  The Theme Muster               - 3 years after .2
#  .4  Hagia Sophia's Bells           - 4 years after .2 (5 after formation)
#  .5  The Varangian Offer            - 7-14 years after .2 (8-15 after formation)
#  .6  The City of the World's Desire - capstone, yearly pulse
#  .7  The Strategos Revolts          - 15-25 years after .3 option B, or Thematic Generals risk
#  .8  Imperial Mismanagement         - Basileus Autokrator + weak administrator
#  .9  Foreign Scholars               - Second Academy
#  .10 The Iconoclast Controversy     - rare, Path B immune
# ==========================================================================

# 1 - The Purple Chamber
flavor_EASRE.1 = {
	type = country_event
	trigger = {
		NOT = { has_variable = fnr_seen_flavor_EASRE_1 } # FNR: once per country (was fire_only_once)
	}
	title = flavor_EASRE.1.title
	desc = flavor_EASRE.1.desc
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		hidden_effect = { set_variable = fnr_seen_flavor_EASRE_1 }
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }
	}

	option = {
		name = flavor_EASRE.1.a
		add_country_modifier = {
			modifier = fnr_EASRE_heir_of_justinian
			years = 10
			mode = replace
		}
		trigger_event_non_silently = { id = flavor_EASRE.2 days = 365 }
	}
}

# 2 - The Patriarch's Question (path select)
flavor_EASRE.2 = {
	type = country_event
	title = flavor_EASRE.2.title
	desc = flavor_EASRE.2.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = EASRE
		NOT = { has_variable = fnr_EASRE_path_orthodoxy }
		NOT = { has_variable = fnr_EASRE_path_hellenistic }
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	# A - Orthodoxy Triumphant
	option = {
		name = flavor_EASRE.2.a
		ai_chance = { factor = 60 }
		set_variable = fnr_EASRE_path_orthodoxy
		add_country_modifier = {
			modifier = fnr_EASRE_orthodoxy_triumphant
			years = 20
			mode = replace
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		custom_tooltip = fnr_EASRE_path_orthodoxy_tt
		fnr_EASRE_create_pentarchy_effect = yes
		hidden_effect = {
			trigger_event_non_silently = { id = flavor_EASRE.3 days = 1095 }
			trigger_event_non_silently = { id = flavor_EASRE.4 days = 1460 }
			trigger_event_non_silently = { id = flavor_EASRE.5 days = { 2555 5110 } }
		}
	}

	# B - Hellenistic Renaissance
	option = {
		name = flavor_EASRE.2.b
		ai_chance = { factor = 40 }
		set_variable = fnr_EASRE_path_hellenistic
		add_country_modifier = {
			modifier = fnr_EASRE_hellenistic_renaissance
			years = 20
			mode = replace
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_weak_penalty
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
		custom_tooltip = fnr_EASRE_path_hellenistic_tt
		fnr_EASRE_create_pentarchy_effect = yes
		hidden_effect = {
			trigger_event_non_silently = { id = flavor_EASRE.3 days = 1095 }
			trigger_event_non_silently = { id = flavor_EASRE.4 days = 1460 }
			trigger_event_non_silently = { id = flavor_EASRE.5 days = { 2555 5110 } }
		}
	}

# 4 - Hagia Sophia's Bells
flavor_EASRE.4 = {
	type = country_event
	title = flavor_EASRE.4.title
	desc = flavor_EASRE.4.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = EASRE
		owns = location:constantinople
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	# A - Full restoration
	option = {
		name = flavor_EASRE.4.a
		trigger = { gold >= 800 }
		add_gold = -800
		add_prestige = prestige_severe_bonus
		location:constantinople = {
			add_location_modifier = { modifier = fnr_EASRE_hagia_sophia_restored years = -1 }
		}
		add_country_modifier = {
			modifier = fnr_EASRE_hagia_sophia_rededicated
			years = 20
			mode = replace
		}
	}

	# B - Basic maintenance
	option = {
		name = flavor_EASRE.4.b
		trigger = { gold >= 200 }
		add_gold = -200
		add_prestige = prestige_weak_bonus
	}

	# C - Path A: Reconsecrate
	option = {
		name = flavor_EASRE.4.c
		trigger = {
			has_variable = fnr_EASRE_path_orthodoxy
			gold >= 800
		}
		add_gold = -800
		add_prestige = prestige_severe_bonus
		add_religious_influence = religious_influence_extreme_bonus
		location:constantinople = {
			add_location_modifier = { modifier = fnr_EASRE_hagia_sophia_restored years = -1 }
		}
		add_country_modifier = {
			modifier = fnr_EASRE_hagia_sophia_rededicated
			years = 20
			mode = replace
		}
	}

	# D - Path B: Public library
	option = {
		name = flavor_EASRE.4.d
		trigger = {
			has_variable = fnr_EASRE_path_hellenistic
			gold >= 800
		}
		add_gold = -800
		add_prestige = prestige_severe_bonus
		location:constantinople = {
			add_location_modifier = { modifier = fnr_EASRE_hagia_sophia_restored years = -1 }
			add_location_modifier = { modifier = fnr_EASRE_hagia_sophia_library years = -1 }
		}
		add_country_modifier = {
			modifier = fnr_EASRE_hagia_sophia_rededicated
			years = 20
			mode = replace
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_weak_penalty
		}
	}

	# E - We cannot afford it
	option = {
		name = flavor_EASRE.4.e
		fallback = yes
		add_prestige = prestige_weak_penalty
	}
}

# 7 - The Strategos Revolts
flavor_EASRE.7 = {
	type = country_event
	title = flavor_EASRE.7.title
	desc = flavor_EASRE.7.desc
	outcome = negative

	trigger = {
		tag = EASRE
		OR = {
			has_variable = fnr_EASRE_strategoi_empowered
			has_estate_privilege = estate_privilege:fnr_EASRE_thematic_generals
		}
	}

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	# A - Buy his loyalty
	option = {
		name = flavor_EASRE.7.a
		ai_chance = { factor = 50 }
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -12
		}
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	# B - Crush him and rein in the themes
	option = {
		name = flavor_EASRE.7.b
		ai_chance = { factor = 50 }
		add_stability = stability_severe_penalty
		add_army_tradition = army_tradition_mild_bonus
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_severe_penalty
		}
		if = {
			limit = { has_variable = fnr_EASRE_strategoi_empowered }
			remove_variable = fnr_EASRE_strategoi_empowered
			remove_country_modifier = fnr_EASRE_empowered_strategoi
		}
	}
}

			mode = replace
		}
	}
}

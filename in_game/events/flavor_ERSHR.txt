# ============================================================
# 1 — Start event (non-silent, fires on_start)
# The restoration is proclaimed; the old faithful recoil.
# ============================================================
flavor_ERSHR.1 = { 
	type = country_event
	title = flavor_ERSHR.1.title
	desc = flavor_ERSHR.1.desc
	fire_only_once = yes
	outcome = negative
	picture = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	immediate = {
		# The world reacts to a Zoroastrian crown ruling Muslim subjects
		every_owned_location = {
			limit = { NOT = { is_capital = yes } }
			change_control = -10
			every_pop = {
				limit = { NOT = { religion = religion:zoroastrian } }
				add_pop_satisfaction = -0.1
			}
		}
	}

	option = {
		name = flavor_ERSHR.1.a
		add_stability = -1
		add_war_exhaustion = 5
	}
	
	trigger = {
		has_variable = fnr_zoroastrian_restored_muslim_reaction
	}

	dynamic_historical_event = {# The time window for the event to occur, only tags specified in it can get the event
		tag = ERSHR		# Can have more than one tag in it
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 100
	}
}

# ============================================================
# 2 — Local uprising
# A pretender prince raises a real rebellion in a restless location.
# ============================================================
flavor_ERSHR.2 = { 
	type = country_event
	title = flavor_ERSHR.2.title
	desc = flavor_ERSHR.2.desc
	fire_only_once = yes
	outcome = negative
	picture = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	immediate = {
		# Create the rebel faction (note: category goes inside data = {})
		create_rebel = {
			name = rebellious_prince_name
			category = pretender
			save_scope_as = brother_rebel
		}
		# Pick a disloyal location, hand its non-Zoroastrian pops to the rebel,
		# and crater control so the rebellion actually grows.
		random_owned_location = {
			save_scope_as = uprising_location
			change_control = -25
			every_pop = {
				limit = { NOT = { religion = religion:zoroastrian } }
				change_pop_allegiance = scope:brother_rebel
			}
		}
		scope:brother_rebel = { add_rebel_progress = 25 }
	}

	option = {	# Send troops — costly suppression that buys back the location
		name = flavor_ERSHR.2.a
		add_yearly_gold = -0.25
		add_manpower = -0.5
		add_war_exhaustion = 3
		scope:brother_rebel = { 
			add_rebel_progress = -0.25 
		}
		scope:uprising_location = { change_control = 0.15 }
	}
	option = {	# Let the governor handle it — cheap now, the region rots
		name = flavor_ERSHR.2.b
		add_stability = -1
		scope:brother_rebel = { add_rebel_progress = 0.15 }
		scope:uprising_location = {
			change_control = -0.10
			change_prosperity = -0.10
		}
	}
	
	trigger = {
		#disaster_type = disaster_type:fnr_ERSHR_zoroastrian_reaction
		any_active_disaster = { disaster_type = disaster_type:fnr_ERSHR_zoroastrian_reaction }
		had_disaster_for_years = {
			disaster_type = disaster_type:fnr_ERSHR_zoroastrian_reaction
			years >= 1
		}
	}

	dynamic_historical_event = {# The time window for the event to occur, only tags specified in it can get the event
		tag = ERSHR		# Can have more than one tag in it
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 1
	}
}

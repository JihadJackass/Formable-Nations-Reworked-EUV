# The Sacred Fires
# Adur Farnbag (the priests' fire). Generally placed in Pars (Fars)
# Adur Gushnasp (the warriors'/kings' fire). This is the one with a solid, archaeologically confirmed location: Takht-e Soleyman (Shiz) in Azerbaijan
# Adur Burzen-Mihr (the farmers' fire). Located in Parthia, in the northeast, traditionally placed on Mount Revand near Nishapur in Khorasan

flavor_ERSHR.14 = { 
	type = country_event
	title = flavor_ERSHR.14.title
	desc = flavor_ERSHR.14.desc
	fire_only_once = yes
	outcome = negative
	picture = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	trigger = {
		owns = location:kariyan
		location:kariyan = {
			religion_percentage = {
				religion = religion:zoroastrian
				value >= 0.10
			}
		}
		owns = location:takht_e_soleyman
		location:takht_e_soleyman = {
			religion_percentage = {
				religion = religion:zoroastrian
				value >= 0.10
			}
		}
		owns = location:nishapur
		location:nishapur = {
			religion_percentage = {
				religion = religion:zoroastrian
				value >= 0.10
			}
		}
	}

	option = { # The Fire is a shining blaze!
		name = flavor_ERSHR.14.a
		add_stability = 10
		add_legitimacy = 20
		religion:sunni = {
			set_religious_view = {
				target = religion:zoroastrian
				value = enemy
			}
		}
		religion:shia = {
			set_religious_view = {
				target = religion:zoroastrian
				value = enemy
			}
		}
	}

	dynamic_historical_event = {# The time window for the event to occur, only tags specified in it can get the event
		tag = ERSHR		# Can have more than one tag in it
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = -100
	}
}

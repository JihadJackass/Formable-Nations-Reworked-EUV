# Pax Romana - The Ptolemaic Kingdom: the House of Ptolemy

# 1 - The House of Ptolemy
fnr_PTLMY.1 = {
	type = country_event
	title = fnr_PTLMY.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = fnr_PTLMY_origin_bactrian }
				desc = fnr_PTLMY.1.desc.bactrian
			}
			triggered_desc = {
				trigger = { has_variable = fnr_PTLMY_origin_egyptian }
				desc = fnr_PTLMY.1.desc.egyptian
			}
			triggered_desc = {
				trigger = { has_variable = fnr_PTLMY_origin_greek }
				desc = fnr_PTLMY.1.desc.greek
			}
			triggered_desc = {
				desc = fnr_PTLMY.1.desc.greek
			}
		}
	}
	fire_only_once = yes
	outcome = positive

	trigger = {
		has_variable = fnr_PTLMY_formed
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
		if = {
			limit = { has_variable = fnr_PTLMY_origin_bactrian }
			add_country_modifier = { modifier = fnr_PTLMY_origin_bactrian years = -1 }
		}
		else_if = {
			limit = { has_variable = fnr_PTLMY_origin_egyptian }
			add_country_modifier = { modifier = fnr_PTLMY_origin_egyptian years = -1 }
		}
		else = {
			add_country_modifier = { modifier = fnr_PTLMY_origin_greek years = -1 }
		}
	}

	option = {
		name = fnr_PTLMY.1.a
		ai_chance = { factor = 50 }
		fnr_PTLMY_change_crown_effect = { VALUE = 15 TT = greek_15 }
		custom_tooltip = fnr_PTLMY_systems_tt
	}

	option = {
		name = fnr_PTLMY.1.b
		ai_chance = { factor = 50 }
		fnr_PTLMY_change_crown_effect = { VALUE = -15 TT = egyptian_15 }
		custom_tooltip = fnr_PTLMY_systems_tt
	}

	option = {
		name = fnr_PTLMY.1.c
		ai_chance = { factor = 50 }
		add_legitimacy = legitimacy_mild_bonus
		custom_tooltip = fnr_PTLMY_systems_tt
	}
}

# 2 - Restoring the House of Ptolemy
fnr_PTLMY.2 = {
	type = country_event
	title = fnr_PTLMY.2.title
	desc = fnr_PTLMY.2.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		owns = location:alexandria
		NOT = { tag_exists = PTLMY }
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:clergy_estate }
	}

	option = {
		name = fnr_PTLMY.2.a
		ai_chance = { factor = 50 }
		custom_tooltip = fnr_PTLMY_bactrian_restoration_tt
		hidden_effect = {
	location:alexandria = {
		create_country_from_location = {
			save_scope_as = fnr_PTLMY_new_egypt
			set_country_rank = country_rank:rank_kingdom
		}
	}
	if = {
		limit = { exists = scope:fnr_PTLMY_new_egypt }
		every_owned_location = {
			limit = {
				OR = {
					area = area:lower_egypt_area
					area = area:upper_egypt_area
					area = area:cyrenaica_area
				}
				is_capital = no
			}
			change_location_owner = scope:fnr_PTLMY_new_egypt
			change_integration_level = core
		}
		scope:fnr_PTLMY_new_egypt = {
			set_variable = fnr_PTLMY_origin_bactrian
			form_country = formable_country:PTLMY_f
			change_country_color = map_PTLMY
			create_character = {
				save_scope_as = fnr_PTLMY_new_ruler
				culture = culture:greek_culture
				religion = religion:fnr_greco_buddhism_religion
			}
			set_new_ruler = scope:fnr_PTLMY_new_ruler
			change_religion = religion:fnr_greco_buddhism_religion
			make_subject_of = {
				target = root
				type = subject_type:vassal
			}
		}
	}
}
	}
}

namespace = fnr_GOTIA

# ==========================================================================
# Pax Romana - Gothia: "The Gothic Kingdom"
#  .1  Crimean Refuge          - on formation (GOTIA_f form_effect)
#  .2  The Bible of Wulfila    - 2-4 years after .1: optional Gothic Arian church
#  .3  Genoese Neighbors       - 4-6 years after .1, someone else must hold Kaffa
#  .4  Gothic Renaissance      - yearly pulse, capstone advance
# ==========================================================================

# 1 - Crimean Refuge
fnr_GOTIA.1 = {
	type = country_event
	title = fnr_GOTIA.1.title
	desc = fnr_GOTIA.1.desc
	fire_only_once = yes
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		hidden_effect = {
			trigger_event_non_silently = { id = fnr_GOTIA.2 days = { 730 1460 } }
			trigger_event_non_silently = { id = fnr_GOTIA.3 days = { 1460 2190 } }
		}
	}

	# A - Fortify the hills
	option = {
		name = fnr_GOTIA.1.a
		ai_chance = { factor = 60 }
		add_country_modifier = { modifier = fnr_GOTIA_hill_forts years = 20 mode = add_and_extend }
	}

	# B - Pay tribute to the steppe
	option = {
		name = fnr_GOTIA.1.b
		trigger = { gold >= 150 }
		ai_chance = { factor = 40 }
		add_gold = -150
		add_stability = stability_weak_bonus
		add_country_modifier = { modifier = fnr_GOTIA_steppe_peace years = 15 mode = add_and_extend }
	}
}

# 2 - The Bible of Wulfila
fnr_GOTIA.2 = {
	type = country_event
	title = fnr_GOTIA.2.title
	desc = fnr_GOTIA.2.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = GOTIA
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	# A - Restore the Arian church of the Goths
	option = {
		name = fnr_GOTIA.2.a
		ai_chance = { factor = 25 }
		change_religion = religion:fnr_gothic_arian_religion
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:fnr_gothic_arian_religion }
		add_stability = stability_mild_penalty
		custom_tooltip = fnr_GOTIA_arian_church_tt
		hidden_effect = {
			every_pop = {
				limit = {
					owner = root
					NOT = { religion = religion:fnr_gothic_arian_religion }
				}
				if = {
					limit = { culture = culture:gothic_culture }
					split_pop = {
						fraction = 0.40
						religion = religion:fnr_gothic_arian_religion
					}
				}
				else = {
					split_pop = {
						fraction = 0.10
						religion = religion:fnr_gothic_arian_religion
					}
				}
			}
		}
	}

	# B - Gothic scripture, Orthodox faith
	option = {
		name = fnr_GOTIA.2.b
		ai_chance = { factor = 50 }
		add_country_modifier = { modifier = fnr_GOTIA_gothic_scripture years = 25 mode = add_and_extend }
	}

	# C - Let the old tongue fade
	option = {
		name = fnr_GOTIA.2.c
		ai_chance = { factor = 25 }
		add_country_modifier = { modifier = fnr_GOTIA_hellenized years = 25 mode = add_and_extend }
	}
}

# 3 - Genoese Neighbors
fnr_GOTIA.3 = {
	type = country_event
	title = fnr_GOTIA.3.title
	desc = fnr_GOTIA.3.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = GOTIA
		location:kaffa.owner ?= {
			NOT = { this = root }
		}
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
		location:kaffa.owner = { save_scope_as = fnr_GOTIA_kaffa_owner }
	}

	# A - A trade compact
	option = {
		name = fnr_GOTIA.3.a
		ai_chance = { factor = 50 }
		add_country_modifier = { modifier = fnr_GOTIA_genoese_trade years = 25 mode = add_and_extend }
		scope:fnr_GOTIA_kaffa_owner ?= {
			add_opinion = { target = root modifier = fnr_GOTIA_genoese_compact }
		}
	}

	# B - Kaffa is Gothic land
	option = {
		name = fnr_GOTIA.3.b
		ai_chance = { factor = 25 }
		add_casus_belli = {
			target = scope:fnr_GOTIA_kaffa_owner
			type = casus_belli:cb_conquer_province
			province = location:kaffa.province
		}
		add_prestige = prestige_weak_penalty
	}

	# C - Hire Genoese crossbowmen
	option = {
		name = fnr_GOTIA.3.c
		trigger = { gold >= 200 }
		ai_chance = { factor = 25 }
		add_gold = -200
		add_country_modifier = { modifier = fnr_GOTIA_genoese_crossbowmen years = 15 mode = add_and_extend }
	}
}

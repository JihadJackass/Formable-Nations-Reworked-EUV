namespace = fnr_RMNRP

# ==========================================================================
# Pax Romana - Roman Republic: "Senatus Populusque Romanus"
#  .1  SPQR Restored              - on formation (RMNRP_f form_effect); starts Senate Authority at 60
#  .2  The Senate Convenes        - 1 year after .1: Optimates or Populares
#  .3  The Consuls Take Office    - every Consular Election (heir selection succession_effect)
#  .4  The Land Law               - 8-15 years after .1 (the Gracchi)
#  .5  The Ambitious General      - at war with a 60+ military general (yearly pulse, 15-year cooldown)
#  .6  Lay Down the Fasces        - dictatorship ended or expired (yearly pulse)
#  .7  Imperator                  - 5 years after refusing to resign: crown yourself or restore the Republic
#  .8  Crossing the Rubicon       - from .5 B, or Senate Authority below 20 (yearly pulse)
#  .9  The Social War             - the Socii demand citizenship (yearly pulse, 2+ members)
#  .10 Res Publica Aeterna        - capstone advance (yearly pulse)
#  .11 Murmurs in the Ludus       - Servile Discontent 60+ (yearly pulse)
#  .12 The New Spartacus          - Servile Discontent 80+ (yearly pulse): the Servile War
#  .13 The End of the Servile War - 3-5 years after .12 A
#  .14 Freedmen Prosper           - Manumission policy (yearly pulse)
#  .15 The Roman Civil Wars       - start of the disaster (fnr_RMNRP_roman_civil_wars)
#  .16 Proscription Lists         - disaster monthly
#  .17 The Legions Choose Sides   - disaster monthly
#  .18 March on Rome              - disaster monthly
#  .20 The Tribune of the Roman People - rare (~3%/yr): a republic holding Rome may restore the Republic early
#  .19 After the Rubicon          - disaster end, when no dictator refused to resign (else .7 fires)
# ==========================================================================

# 1 - SPQR Restored
fnr_RMNRP.1 = {
	type = country_event
	title = fnr_RMNRP.1.title
	desc = fnr_RMNRP.1.desc
	fire_only_once = yes
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		set_variable = { name = fnr_RMNRP_senate_authority value = 60 }
		fnr_RMNRP_refresh_senate_tier_effect = yes
		hidden_effect = {
			trigger_event_non_silently = { id = fnr_RMNRP.2 days = 365 }
			trigger_event_non_silently = { id = fnr_RMNRP.4 days = { 2920 5475 } }
		}
	}

	# A - The Senate is sovereign
	option = {
		name = fnr_RMNRP.1.a
		ai_chance = { factor = 50 }
		fnr_RMNRP_change_senate_effect = { VALUE = 15 }
		add_stability = stability_mild_bonus
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		custom_tooltip = fnr_RMNRP_senate_plus_15_tt
	}

	# B - The Senate and the People
	option = {
		name = fnr_RMNRP.1.b
		ai_chance = { factor = 50 }
		add_republican_tradition = republican_tradition_mild_bonus
		add_estate_satisfaction = {
			type = estate_type:peasants_estate
			value = estate_satisfaction_mild_bonus
		}
		add_country_modifier = { modifier = fnr_RMNRP_senate_and_people years = 20 mode = add_and_extend }
	}
}

fnr_RMNRP.8 = {
	type = country_event
	title = fnr_RMNRP.8.title
	desc = fnr_RMNRP.8.desc
	outcome = negative

	trigger = {
		has_or_had_tag = RMNRP
		government_type = government_type:republic
		NOT = { has_variable = fnr_RMNRP_rubicon_cooldown }
	}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		set_variable = { name = fnr_RMNRP_rubicon_cooldown years = 20 }
	}

	option = {
		name = fnr_RMNRP.8.a
		fnr_RMNRP_change_senate_effect = { VALUE = -10 }
		add_stability = stability_mild_penalty
		fnr_RMNRP_civil_war_effect = { NAME = fnr_RMNRP_rubicon_rebels }
		if = {
			limit = {
				exists = scope:fnr_RMNRP_rebel
				has_variable = fnr_RMNRP_rubicon_general
				var:fnr_RMNRP_rubicon_general ?= { is_alive = yes }
			}
			var:fnr_RMNRP_rubicon_general ?= {
				change_character_allegiance = scope:fnr_RMNRP_rebel
			}
			set_variable = { name = fnr_RMNRP_rebel_leader value = var:fnr_RMNRP_rubicon_general }
		}
		if = {
			limit = { has_variable = fnr_RMNRP_rubicon_general }
			remove_variable = fnr_RMNRP_rubicon_general
		}
		custom_tooltip = fnr_RMNRP_rubicon_tt
	}
}

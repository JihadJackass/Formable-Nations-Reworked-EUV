# The Path Forward
flavor_RHLND.1 = {
	type = country_event
	title = flavor_RHLND.1.title
	desc = flavor_RHLND.1.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = RHLND
	}

	image = "gfx/interface/illustrations/event/backgrounds/exterior/nobles/north_german_nobles_exterior.dds"

	option = { # Das Rheinische Königreich
		name = flavor_RHLND.1.a
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		change_government_type = government_type:monarchy
		if = {
			limit = { has_reform = government_reform:fnr_GSCTL_rheinkonfoderation }
			remove_reform = government_reform:fnr_GSCTL_rheinkonfoderation
		}
		revoke_estate_privilege = estate_privilege:fnr_RHLND_squabbles_of_the_nobility
		grant_estate_privilege = estate_privilege:fnr_RHLND_cooperative_nobles
		add_reform = government_reform:fnr_GSCTL_das_rheinische_konigreich
	}
	option = { # Der Rheinbund
		name = flavor_RHLND.1.b
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		change_government_type = government_type:republic
		if = {
			limit = { has_reform = government_reform:fnr_GSCTL_rheinkonfoderation }
			remove_reform = government_reform:fnr_GSCTL_rheinkonfoderation
		}
		revoke_estate_privilege = estate_privilege:fnr_RHLND_squabbles_of_the_nobility
		grant_estate_privilege = estate_privilege:fnr_RHLND_cooperative_nobles
		add_reform = government_reform:fnr_GSCTL_der_rheinbund
	}
}

# Die Wacht am Rhein
flavor_RHLND.2 = {
	type = country_event
	title = flavor_RHLND.2.title
	desc = flavor_RHLND.2.desc
	fire_only_once = yes
	outcome = neutral

	# Triggers after 10 years (120 months) at war in the Age of Absolutism or later.
	# The months are counted by on_action/fnr_RHLND_pulses.txt (fnr_RHLND_war_months).
	trigger = {
		tag = RHLND
		current_age_or_later = { age = age_5_absolutism }
		trigger_if = {
			limit = { has_variable = fnr_RHLND_war_months }
			var:fnr_RHLND_war_months >= 120
		}
		trigger_else = { always = no }
	}

	immediate = {
		# sets the required variables to meet the rheinwachtgesetz recruitment law, once NOT_MET has expired you will be able to pass the law
		set_variable = { 
			name = fnr_has_die_wacht_am_rhein_10years_NOT_MET
			value = yes
			days = 3650
		}
		set_variable = fnr_has_die_wacht_am_rhein_10years_HAS_MET	# permanent
	}

	option = {
		name = flavor_RHLND.2.a
		add_reform = government_reform:fnr_GSCTL_die_wacht_am_rhein
	}
}

# The Pillar that is the Church
# Event to replace High Church Influence with The Pillar that is the Church 
# Revoked by hand: fnr_high_church_influence_HAS_removed (the clergy are angered, no Pillar; see the privilege's on_deactivate)
# Replaced by this event: fnr_high_church_influence_removed
flavor_RHLND.4 = {
	type = country_event
	title = flavor_RHLND.4.title
	desc = flavor_RHLND.4.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = RHLND
		# Still in place: replaced once the clergy is weak, or after 50 years (fnr_high_church_influence_NOT_removed
		# is set for 50 years at formation). A privilege revoked by hand was already replaced by The Pillar.
		has_estate_privilege = estate_privilege:fnr_RHLND_high_church_influence
		NOT = { has_estate_privilege = estate_privilege:fnr_RHLND_pillar_that_is_the_church }
		OR = {
			NOT = { has_variable = fnr_high_church_influence_NOT_removed }
			AND = {
				any_estate = {
					estate_type = estate_type:clergy_estate
					power < 0.10
				}
				state_religion_clergy < 0.10
			}
		}
	}

	option = {
		name = flavor_RHLND.4.a
		set_variable = fnr_high_church_influence_removed
		revoke_estate_privilege = estate_privilege:fnr_RHLND_high_church_influence
		grant_estate_privilege = estate_privilege:fnr_RHLND_pillar_that_is_the_church
	}
}

		}
	}
}

# 7 - A member state seeks to leave
flavor_RHLND.7 = {
	type = country_event
	title = flavor_RHLND.7.title
	desc = flavor_RHLND.7.desc
	outcome = negative

	trigger = {
		tag = RHLND
		any_owned_location = { is_capital = no }
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
		capital ?= { save_scope_as = fnr_RHLND_capital }
		random_owned_location = {
			limit = {
				is_capital = no
				NOT = { scope:fnr_RHLND_capital ?= { province = prev.province } }
			}
			save_scope_as = fnr_RHLND_seceding
		}
		if = {
			limit = { NOT = { exists = scope:fnr_RHLND_seceding } }
			random_owned_location = {
				limit = { is_capital = no }
				save_scope_as = fnr_RHLND_seceding
			}
		}
	}

	option = { # Let them go
		name = flavor_RHLND.7.a
		ai_chance = { factor = 30 }
		custom_tooltip = flavor_RHLND.7.a.tt
		hidden_effect = {
			scope:fnr_RHLND_seceding ?= {
				create_country_from_location = {
					save_scope_as = fnr_RHLND_seceded
					set_country_rank = country_rank:rank_county
				}
			}
			if = {
				limit = { exists = scope:fnr_RHLND_seceded }
				scope:fnr_RHLND_seceding.province ?= {
					every_location_in_province = {
						limit = { owner = root }
						change_location_owner = scope:fnr_RHLND_seceded
					}
				}
			}
		}
		add_stability = stability_weak_bonus
	}

	option = { # Hold them by force
		name = flavor_RHLND.7.b
		ai_chance = { factor = 70 }
		add_stability = stability_mild_penalty
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}
}

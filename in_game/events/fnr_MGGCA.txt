# ---------- Rome, Byzantium and the Ptolemies ----------

# 20 - The Shadow of Rome (once)
fnr_MGGCA.20 = {
	type = country_event
	title = fnr_MGGCA.20.title
	desc = fnr_MGGCA.20.desc
	outcome = neutral
	fire_only_once = yes

	trigger = {
		has_or_had_tag = MGGCA
		OR = {
			exists = c:ROM
			exists = c:RMNRP
		}
		NOT = { has_country_modifier = fnr_MGGCA_friend_of_rome }
		NOT = { has_country_modifier = fnr_MGGCA_defiance_of_rome }
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
	}

	option = {
		name = fnr_MGGCA.20.a
		ai_chance = { factor = 60 }
		add_country_modifier = { modifier = fnr_MGGCA_friend_of_rome years = 25 }
	}

	option = {
		name = fnr_MGGCA.20.b
		ai_chance = { factor = 40 }
		add_country_modifier = { modifier = fnr_MGGCA_defiance_of_rome years = 25 }
		add_prestige = prestige_weak_bonus
	}
}

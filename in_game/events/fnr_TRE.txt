# 2 - Turkmen Raid (Peril 50+)
fnr_TRE.2 = {
	type = country_event
	title = fnr_TRE.2.title
	desc = fnr_TRE.2.desc
	outcome = negative

	trigger = {
		fnr_TRE_is_komnenian = yes
		fnr_TRE_has_peril = yes
		fnr_TRE_peril_at_least = { VALUE = 50 }
		any_neighbor_country = { fnr_TRE_is_turkish = yes }
	}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
	}

	# Ride out to meet them
	option = {
		name = fnr_TRE.2.a
		ai_chance = { factor = 40 }
		add_manpower = -0.5
		add_army_tradition = army_tradition_weak_bonus
		fnr_TRE_change_peril_effect = { VALUE = -5 TT = minus_5 }
	}

	# Buy them off
	option = {
		name = fnr_TRE.2.b
		ai_chance = { factor = 40 }
		trigger = { gold >= 50 }
		add_gold = -50
	}

	# Shelter behind the walls
	option = {
		name = fnr_TRE.2.c
		ai_chance = { factor = 20 }
		add_country_modifier = { modifier = fnr_TRE_raid_damage years = 3 }
		fnr_TRE_change_peril_effect = { VALUE = 5 TT = plus_5 }
	}
}

# 3 - The Sultan's Army Before Trebizond (Peril 80+)
fnr_TRE.3 = {
	type = country_event
	title = fnr_TRE.3.title
	desc = fnr_TRE.3.desc
	outcome = negative

	trigger = {
		fnr_TRE_is_komnenian = yes
		fnr_TRE_peril_at_least = { VALUE = 80 }
		any_neighbor_country = { fnr_TRE_is_turkish = yes }
	}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
		set_variable = { name = fnr_TRE_siege_cooldown years = 10 }
		random_neighbor_country = {
			limit = { fnr_TRE_is_turkish = yes }
			save_scope_as = fnr_TRE_sultan
		}
	}

	# Pay what he asks
	option = {
		name = fnr_TRE.3.a
		ai_chance = { factor = 40 }
		trigger = { gold >= 250 }
		add_gold = -250
		add_prestige = prestige_mild_penalty
		fnr_TRE_change_peril_effect = { VALUE = -25 TT = minus_25 }
	}

	# Defy him from the walls
	option = {
		name = fnr_TRE.3.b
		ai_chance = { factor = 30 }
		add_stability = stability_weak_penalty
		add_country_modifier = { modifier = fnr_TRE_defiance years = 10 }
		fnr_TRE_change_peril_effect = { VALUE = -10 TT = minus_10 }
	}

	# Offer him a Komnenian bride
	option = {
		name = fnr_TRE.3.c
		ai_chance = { factor = 30 }
		trigger = {
			fnr_TRE_pacts_below = { VALUE = 3 }
			scope:fnr_TRE_sultan ?= { NOT = { has_variable = fnr_TRE_bride_of_trebizond } }
		}
		add_prestige = prestige_weak_penalty
		scope:fnr_TRE_sultan ?= {
			set_variable = { name = fnr_TRE_bride_of_trebizond years = 30 }
			add_opinion = { modifier = fnr_TRE_komnenian_kin target = root }
		}
		scope:fnr_TRE_sultan ?= {
			root = { add_opinion = { modifier = fnr_TRE_komnenian_kin target = prev } }
		}
		custom_tooltip = fnr_TRE_bride_pact_tt
		fnr_TRE_change_peril_effect = { VALUE = -25 TT = minus_25 }
		fnr_TRE_count_pacts_effect = yes
	}
}

# 4 - A Broken Pact (a marriage partner is at war with us)
fnr_TRE.4 = {
	type = country_event
	title = fnr_TRE.4.title
	desc = fnr_TRE.4.desc
	outcome = negative

	trigger = {
		fnr_TRE_is_komnenian = yes
		any_country = {
			has_variable = fnr_TRE_bride_of_trebizond
			is_at_war_with = root
		}
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
		every_country = {
			limit = {
				has_variable = fnr_TRE_bride_of_trebizond
				is_at_war_with = root
			}
			remove_variable = fnr_TRE_bride_of_trebizond
			root = { add_opinion = { modifier = fnr_TRE_broken_pact target = prev } }
		}
		fnr_TRE_count_pacts_effect = yes
	}

	option = {
		name = fnr_TRE.4.a
		ai_chance = { factor = 100 }
		add_prestige = prestige_mild_penalty
		fnr_TRE_change_peril_effect = { VALUE = 10 TT = plus_10 }
	}
}

# 8 - The Two Emperors (Byzantium or the Eastern Roman Empire)
fnr_TRE.8 = {
	type = country_event
	title = fnr_TRE.8.title
	desc = fnr_TRE.8.desc
	outcome = neutral
	fire_only_once = yes

	trigger = {
		fnr_TRE_is_komnenian = yes
		NOT = { fnr_TRE_is_byzantium = yes }
		any_country = { fnr_TRE_is_byzantium = yes }
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:clergy_estate }
		random_country = {
			limit = { fnr_TRE_is_byzantium = yes }
			save_scope_as = fnr_TRE_byzantium
		}
	}

	# We are the true Emperors
	option = {
		name = fnr_TRE.8.a
		ai_chance = { factor = 50 }
		add_country_modifier = { modifier = fnr_TRE_rival_emperors years = 25 }
		scope:fnr_TRE_byzantium ?= {
			add_opinion = { modifier = fnr_TRE_rival_emperor target = root }
		}
	}

	# One family, two thrones
	option = {
		name = fnr_TRE.8.b
		ai_chance = { factor = 50 }
		add_country_modifier = { modifier = fnr_TRE_komnenian_reconciliation years = 25 }
		scope:fnr_TRE_byzantium ?= {
			add_opinion = { modifier = fnr_TRE_komnenian_kinship target = root }
		}
		scope:fnr_TRE_byzantium ?= {
			root = { add_opinion = { modifier = fnr_TRE_komnenian_kinship target = prev } }
		}
	}
}
